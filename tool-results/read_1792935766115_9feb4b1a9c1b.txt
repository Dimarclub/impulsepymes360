   489→// ===================== PLANO LIVE =====================
   490→function PlanoLive({ data, loading, onBuy }: { data: BagPlanData | null; loading: boolean; onBuy: () => void }) {
   491→  const [view, setView] = useState<'current' | 'next'>('current')
   492→
   493→  const edition = view === 'current' ? data?.currentEdition : data?.nextEdition
   494→  const spaces = view === 'current' ? data?.currentSpaces : data?.nextSpaces
   495→  const stats = view === 'current' ? data?.currentStats : data?.nextStats
   496→
   497→  // Para la campaña actual (todos vendidos), agrupamos sponsors para la leyenda
   498→  const sponsors = useMemo(() => {
   499→    if (!spaces) return []
   500→    const map = new Map<string, number>()
   501→    for (const s of spaces) {
   502→      if (s.status !== 'SOLD') continue
   503→      const n = s.buyerName || ''
   504→      if (!n) continue
   505→      map.set(n, (map.get(n) || 0) + 1)
   506→    }
   507→    return Array.from(map.entries()).sort((a, b) => b[1] - a[1]).slice(0, 12)
   508→  }, [spaces])

   651→function LivePanel({ panel, spaces, view }: { panel: BagPanel; spaces: BagSpace[]; view: 'current' | 'next' }) {
   652→  const byPos = useMemo(() => {
   653→    const m = new Map<number, BagSpace>()
   654→    for (const s of spaces) m.set(s.position, s)
   655→    return m
   656→  }, [spaces])
   657→
   658→  const positions: number[] = []
   659→  for (let i = panel.start; i <= panel.end; i++) positions.push(i)
   660→
   661→  return (
   662→    <div className="rounded-xl p-4" style={{ background: C.card, border: `1px solid ${C.border}` }}>
   663→      <div className="flex items-center justify-between mb-3">
   664→        <div>
   665→          <p style={{ fontSize: 14, fontWeight: 600, color: C.text }}>{panel.label}</p>
   666→          <p style={{ fontSize: 11, color: C.muted }}>{panel.widthCm}×{panel.heightCm} cm · {panel.cols}×{panel.rows}</p>
   667→        </div>
   668→        <span style={{ fontSize: 11, color: C.muted }}>#{panel.start}–{panel.end}</span>
   669→      </div>
   670→      <div
   671→        className="grid"
   672→        style={{ gridTemplateColumns: `repeat(${panel.cols}, minmax(0, 1fr))`, gap: 2 }}
   673→      >
   674→        {positions.map((pos) => {
   675→          const s = byPos.get(pos)
   676→          let bg = C.bg
   677→          let title = `#${pos} · Disponible`
   678→          if (view === 'current') {
   679→            // Todos vendidos → color por sponsor
   680→            const name = s?.buyerName
   681→            if (name) {
   682→              bg = sponsorColor(name)
   683→              title = `#${pos} · ${name}`
   684→            }
   685→          } else {
   686→            // next → color por status
   687→            if (s?.status === 'SOLD') {
   688→              const name = s.buyerName
   689→              bg = name ? sponsorColor(name) : C.rose
   690→              title = `#${pos} · Vendido${name ? ` · ${name}` : ''}`
   691→            } else if (s?.status === 'RESERVED') {
   692→              bg = C.amber
   693→              title = `#${pos} · Reservado`
   694→            } else {
   695→              bg = `${C.emerald}33`
   696→              title = `#${pos} · Disponible`
   697→            }
   698→          }
   699→          return (
   700→            <div
   701→              key={pos}
   702→              title={title}
   703→              style={{
   704→                aspectRatio: '1 / 1',
   705→                background: bg,
   706→                borderRadius: 2,
   707→                minWidth: 6,
   708→                cursor: 'default',
   709→              }}
   710→            />
   711→          )
   712→        })}
   713→      </div>
   714→    </div>
   715→  )
   716→}
