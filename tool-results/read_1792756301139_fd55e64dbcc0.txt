   167→export function BagPlanModule() {
   168→  const [tab, setTab] = useState('plano')
   169→  const [data, setData] = useState<BagPlanData | null>(null)
   170→  const [loading, setLoading] = useState(true)
   171→  const [error, setError] = useState<string | null>(null)
   172→  const [filterPanel, setFilterPanel] = useState<string>('all')
   173→  const [filterType, setFilterType] = useState<string>('all')
   174→  const [search, setSearch] = useState('')
   175→  const [selectedSpace, setSelectedSpace] = useState<BagSpace | null>(null)
   176→  const toast = useAppStore((s) => s.toast)
   177→  const currentUser = useAppStore((s) => s.currentUser)
   178→
   179→  const fetchData = useCallback(async () => {
   180→    setLoading(true)
   181→    setError(null)
   182→    try {
   183→      const res = await fetch('/api/impulse/bag-plan')
   184→      if (!res.ok) throw new Error('Error al cargar')
   185→      const json = (await res.json()) as BagPlanData
   186→      setData(json)
   187→    } catch (e) {
   188→      setError(e instanceof Error ? e.message : 'Error desconocido')
   189→    } finally {
   190→      setLoading(false)
   191→    }
   192→  }, [])
   193→
   194→  useEffect(() => {
   195→    fetchData()
   196→  }, [fetchData])
   197→
   198→  // Filtrado de espacios
   199→  const filteredSpaces = useMemo(() => {
   200→    if (!data) return []
   201→    return data.spaces.filter((s) => {
   202→      if (filterPanel !== 'all') {
   203→        const p = panelForPosition(s.position)
   204→        if (p.key !== filterPanel) return false
   205→      }
   206→      if (filterType !== 'all' && s.type !== filterType) return false
   207→      if (search.trim()) {
   208→        const q = search.toLowerCase().trim()
   209→        const name = s.provider?.companyName?.toLowerCase() || ''
   210→        if (!name.includes(q)) return false
   211→      }
   212→      return true
   213→    })
   214→  }, [data, filterPanel, filterType, search])
   215→
   216→  // Sponsor lookup map
   217→  const sponsors = useMemo(() => {
   218→    if (!data) return []
   219→    const map = new Map<string, { id: string; name: string; logoUrl?: string | null; count: number; invested: number; status: string }>()
   220→    for (const s of data.spaces) {
   221→      if (s.status === 'SOLD' && s.provider) {
   222→        const key = s.provider.companyName
   223→        const existing = map.get(key) || { id: s.providerId || key, name: key, logoUrl: s.provider.logoUrl, count: 0, invested: 0, status: 'Activo' }
   224→        existing.count++
   225→        existing.invested += s.price || 0
   226→        map.set(key, existing)
   227→      }
   228→    }
   229→    return Array.from(map.values()).sort((a, b) => b.invested - a.invested)
   230→  }, [data])
   231→
   232→  const handlePatched = (updated: BagSpace) => {
   233→    if (!data) return
   234→    // Reemplazar el espacio actualizado y recalcular stats
   235→    const next = data.spaces.map((s) => (s.id === updated.id ? { ...s, ...updated } : s))
   236→    let available = 0, reserved = 0, sold = 0, revenue = 0
   237→    for (const s of next) {
   238→      if (s.status === 'AVAILABLE') available++
   239→      else if (s.status === 'RESERVED') reserved++
   240→      else if (s.status === 'SOLD') { sold++; revenue += s.price || 0 }
   241→    }
   242→    setData({
   243→      ...data,
   244→      spaces: next,
   245→      stats: { total: data.stats.total, available, reserved, sold, revenue, potential: data.stats.potential },
   246→    })
   247→    setSelectedSpace(null)
   248→  }
   249→
   250→  if (loading) {
   251→    return (
   252→      <div className="space-y-6 p-4 md:p-6 lg:p-8 max-w-[1320px] mx-auto">
   253→        <Skeleton className="h-20 w-full" />
   254→        <Skeleton className="h-10 w-full max-w-2xl" />
   255→        <Skeleton className="h-64 w-full" />
   256→        <Skeleton className="h-64 w-full" />
   257→      </div>
   258→    )
   259→  }
   260→
   261→  if (error || !data) {
   262→    return (
   263→      <div className="space-y-6 p-4 md:p-6 lg:p-8 max-w-[1320px] mx-auto">
   264→        <Card>
   265→          <CardContent className="py-12 text-center">
   266→            <AlertTriangle className="size-10 mx-auto text-amber-500 mb-3" />
   267→            <p className="font-medium">No se pudo cargar el plano del bolso</p>
   268→            <p className="text-sm text-muted-foreground mt-1">{error || 'Intenta nuevamente'}</p>
   269→            <Button className="mt-4" onClick={fetchData}>
   270→              <RefreshCw className="size-4 mr-2" /> Reintentar
   271→            </Button>
   272→          </CardContent>
   273→        </Card>
   274→      </div>
   275→    )
   276→  }
   277→
   278→  return (
   279→    <div className="space-y-6 p-4 md:p-6 lg:p-8 max-w-[1320px] mx-auto">
   280→      {/* Header */}
   281→      <Card className="overflow-hidden border-emerald-200">
   282→        <CardContent className="p-5 md:p-6">
   283→          <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-4">
   284→            <div className="space-y-1">
   285→              <div className="flex items-center gap-2 flex-wrap">
   286→                <Package className="size-5 text-emerald-600" />
   287→                <h1 className="text-xl md:text-2xl font-bold tracking-tight">{data.edition.name}</h1>
   288→                <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100">
   289→                  {data.edition.status === 'OPEN' ? 'Abierta' : data.edition.status}
   290→                </Badge>
   291→              </div>
   292→              <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-sm text-muted-foreground">
   293→                <span className="inline-flex items-center gap-1"><MapPin className="size-3.5" /> {data.edition.city}</span>
   294→                <span className="inline-flex items-center gap-1"><Calendar className="size-3.5" /> {fmtDate(data.edition.startDate)} – {fmtDate(data.edition.endDate)}</span>
   295→                <span className="inline-flex items-center gap-1"><Layers className="size-3.5" /> {data.edition.printQuantity} bolsas</span>
   296→                <span className="inline-flex items-center gap-1"><Package className="size-3.5" /> Bolso {data.edition.bagSize} cm</span>
   297→              </div>
   298→            </div>
   299→            <Button variant="outline" size="sm" onClick={fetchData}>
   300→              <RefreshCw className="size-4 mr-1.5" /> Actualizar
   301→            </Button>
   302→          </div>
   303→
   304→          {/* Stat pills */}
   305→          <div className="grid grid-cols-2 md:grid-cols-4 gap-3 mt-5">
   306→            <StatPill label="Total espacios" value={data.stats.total} tone="muted" icon={<Package className="size-4" />} />
   307→            <StatPill label="Disponibles" value={data.stats.available} tone="emerald" icon={<CheckCircle2 className="size-4" />} />
   308→            <StatPill label="Reservados" value={data.stats.reserved} tone="amber" icon={<Clock className="size-4" />} />
   309→            <StatPill label="Vendidos" value={data.stats.sold} tone="rose" icon={<ShoppingBag className="size-4" />} />
   310→          </div>
   311→        </CardContent>
   312→      </Card>
   313→
   314→      {/* Tabs */}
   315→      <Tabs value={tab} onValueChange={setTab}>
   316→        <TabsList className="grid w-full grid-cols-2 md:grid-cols-4 max-w-2xl">
   317→          <TabsTrigger value="plano"><Package className="size-4 md:mr-1.5" /><span className="hidden sm:inline">Plano</span></TabsTrigger>
   318→          <TabsTrigger value="mapa"><MapPin className="size-4 md:mr-1.5" /><span className="hidden sm:inline">Mapa</span></TabsTrigger>
   319→          <TabsTrigger value="sponsors"><Store className="size-4 md:mr-1.5" /><span className="hidden sm:inline">Patrocinadores</span></TabsTrigger>
   320→          <TabsTrigger value="dashboard"><TrendingUp className="size-4 md:mr-1.5" /><span className="hidden sm:inline">Dashboard</span></TabsTrigger>
   321→        </TabsList>
   322→
   323→        <TabsContent value="plano" className="mt-4 space-y-4">
   324→          <Toolbar
   325→            filterPanel={filterPanel}
   326→            setFilterPanel={setFilterPanel}
   327→            filterType={filterType}
   328→            setFilterType={setFilterType}
   329→            search={search}
   330→            setSearch={setSearch}
   331→          />
   332→          <BagView
   333→            spaces={filteredSpaces}
   334→            onCellClick={setSelectedSpace}
   335→            currentUserId={currentUser?.id}
   336→            view="status"
   337→          />
   338→          <Legend />
   339→        </TabsContent>
   340→
   341→        <TabsContent value="mapa" className="mt-4 space-y-4">
   342→          <Toolbar
   343→            filterPanel={filterPanel}
   344→            setFilterPanel={setFilterPanel}
   345→            filterType={filterType}
   346→            setFilterType={setFilterType}
   347→            search={search}
   348→            setSearch={setSearch}
   349→          />
   350→          <BagView
   351→            spaces={filteredSpaces}
   352→            onCellClick={setSelectedSpace}
   353→            currentUserId={currentUser?.id}
   354→            view="sponsor"
   355→          />

   357→        </TabsContent>
   358→
   359→        <TabsContent value="sponsors" className="mt-4">
   360→          <SponsorsTab sponsors={sponsors} onPick={(name) => { setSearch(name); setTab('plano'); toast(`Mostrando espacios de ${name}`, 'info') }} />

   368→      {/* Space Editor */}

   472→function BagView({
   473→  spaces,
   474→  onCellClick,
   475→  currentUserId,
   476→  view,
   477→}: {
   478→  spaces: BagSpace[]
   479→  onCellClick: (s: BagSpace) => void
   480→  currentUserId?: string
   481→  view: 'status' | 'sponsor'
   482→}) {
   483→  // Indexar espacios por posición para acceso rápido
   484→  const byPosition = useMemo(() => {
   485→    const m = new Map<number, BagSpace>()
   486→    for (const s of spaces) m.set(s.position, s)
   487→    return m
   488→  }, [spaces])
   489→
   490→  return (
   491→    <div className="space-y-4">
   492→      {BAG_PANELS.map((panel) => {
   493→        const positions: number[] = []
   494→        for (let i = panel.start; i <= panel.end; i++) positions.push(i)
   495→        const panelSpaces = positions.map((p) => byPosition.get(p)).filter(Boolean) as BagSpace[]
   496→        const sold = panelSpaces.filter((s) => s.status === 'SOLD').length
   497→        const available = panelSpaces.filter((s) => s.status === 'AVAILABLE').length
   498→        const total = panel.end - panel.start + 1
   499→
   500→        return (
   501→          <Card key={panel.key}>
   502→            <CardHeader className="pb-3">
   503→              <div className="flex flex-wrap items-center justify-between gap-2">
   504→                <div>
   505→                  <CardTitle className="text-base flex items-center gap-2">
   506→                    {panel.label}
   507→                    <span className="text-xs font-normal text-muted-foreground">
   508→                      · {panel.widthCm}×{panel.heightCm} cm · {panel.cols}×{panel.rows}
   509→                    </span>
   510→                  </CardTitle>
   511→                  <CardDescription className="text-xs">
   512→                    Espacios {panel.start}–{panel.end} · {total} celdas de 5×5 cm
   513→                  </CardDescription>
   514→                </div>
   515→                <div className="flex items-center gap-1.5">
   516→                  <Badge variant="outline" className="bg-emerald-50 text-emerald-700 border-emerald-200">{available} libres</Badge>
   517→                  <Badge variant="outline" className="bg-rose-50 text-rose-700 border-rose-200">{sold} vendidas</Badge>
   518→                </div>
   519→              </div>
   520→            </CardHeader>
   521→            <CardContent>
   522→              <div
   523→                className="grid gap-[2px] sm:gap-1"
   524→                style={{ gridTemplateColumns: `repeat(${panel.cols}, minmax(0, 1fr))` }}
   525→              >
   526→                {positions.map((pos) => {
   527→                  const s = byPosition.get(pos)
   528→                  return (
   529→                    <Cell
   530→                      key={pos}
   531→                      position={pos}
   532→                      space={s}
   533→                      onCellClick={onCellClick}
   534→                      currentUserId={currentUserId}
   535→                      view={view}
   536→                    />
   537→                  )
   538→                })}
   539→              </div>
   540→            </CardContent>
   541→          </Card>
   542→        )
   543→      })}
   544→    </div>
   545→  )
   546→}

   548→function Cell({
   549→  position,
   550→  space,
   551→  onCellClick,
   552→  currentUserId,
   553→  view,
   554→}: {
   555→  position: number
   556→  space?: BagSpace
   557→  onCellClick: (s: BagSpace) => void
   558→  currentUserId?: string
   559→  view: 'status' | 'sponsor'
   560→}) {
   561→  const { row, col } = rowColForPosition(position)
   562→  const status = space?.status || 'AVAILABLE'
   563→  const isMine = !!space?.providerId && space.providerId === currentUserId
   564→
   565→  // Estilos por vista
   566→  let bg = 'bg-muted/40 border-border hover:bg-muted/70'
   567→  let textColor = 'text-muted-foreground'
   568→
   569→  if (view === 'sponsor') {
   570→    if (status === 'SOLD' && space?.provider?.companyName) {
   571→      const c = sponsorColor(space.provider.companyName)
   572→      return (
   573→        <Tooltip>
   574→          <TooltipTrigger asChild>

   939→function SpaceEditorDialog({
   940→  space,
   941→  onClose,
   942→  onPatched,
   943→  currentUser,
   944→  toast,
   945→}: {
   946→  space: BagSpace | null

   948→  onPatched: (s: BagSpace) => void
   949→  currentUser: { id: string; role: string; name: string } | null
   950→  toast: (msg: string, type?: 'success' | 'error' | 'info' | 'warning') => void
   951→}) {
   952→  const [providers, setProviders] = useState<SupplierOption[]>([])
   953→  const [providerId, setProviderId] = useState<string>('')
   954→  const [type, setType] = useState<string>('QR')
   955→  const [imageUrl, setImageUrl] = useState<string>('')
   956→  const [price, setPrice] = useState<string>('20')
   957→  const [saving, setSaving] = useState(false)
   958→  const [providersError, setProvidersError] = useState(false)
   959→
   960→  // Cargar proveedores para el select
   961→  useEffect(() => {
   962→    if (!space) return
   963→    let cancelled = false
   964→    fetch('/api/suppliers')
   965→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error('no-ok'))))
   966→      .then((rows: any[]) => {
   967→        if (cancelled) return
   968→        const opts = rows.map((r) => ({ id: r.id, companyName: r.companyName, logoUrl: r.logoUrl }))
   969→        setProviders(opts)
   970→        setProvidersError(false)
   971→      })
   972→      .catch(() => {
   973→        setProviders([])
   974→        setProvidersError(true)
   975→      })
   976→    return () => { cancelled = true }
   977→  }, [space])
   978→
   979→  // Reset form cuando cambia el space
   980→  useEffect(() => {
   981→    if (space) {
   982→      setProviderId(space.providerId || '')
   983→      setType(space.type || 'QR')
   984→      setImageUrl(space.imageUrl || '')
   985→      setPrice(String(space.price ?? 20))
   986→    }
   987→  }, [space])
   988→
   989→  const { panel, row, col } = space ? rowColForPosition(space.position) : { panel: null, row: 0, col: 0 }
   990→  const isAdmin = currentUser?.role === 'ADMIN'
   991→  const isSupplier = currentUser?.role === 'SUPPLIER'
   992→  const canAssign = isAdmin || isSupplier
   993→  const isReserved = space?.status === 'RESERVED'
   994→  const isSold = space?.status === 'SOLD'
   995→  const isAvailable = space?.status === 'AVAILABLE'
   996→
   997→  const patch = async (action: 'reserve' | 'release' | 'assign') => {
   998→    if (!space) return
   999→    if (!currentUser) {
  1000→      toast('Inicia sesión para gestionar espacios', 'warning')
  1001→      return
  1002→    }
  1003→    setSaving(true)
  1004→    try {
  1005→      const body: any = { spaceId: space.id, action }
  1006→      if (action === 'assign') {
  1007→        if (!providerId) {
  1008→          toast('Selecciona un proveedor', 'warning')
  1009→          setSaving(false)
  1010→          return
  1011→        }
  1012→        body.providerId = providerId
  1013→        body.type = type
  1014→        body.imageUrl = imageUrl || undefined
  1015→        const pNum = parseFloat(price)
  1016→        if (!isNaN(pNum) && pNum >= 0) body.price = pNum
  1017→      }
  1018→      const res = await fetch('/api/impulse/bag-plan', {
  1019→        method: 'PATCH',
  1020→        headers: { 'Content-Type': 'application/json' },
  1021→        body: JSON.stringify(body),
  1022→      })
  1023→      const d = await res.json()
  1024→      if (!res.ok) {
  1025→        toast(d.error || 'Error al actualizar', 'error')
  1026→        return
  1027→      }
  1028→      toast(`Espacio #${space.position} ${action === 'reserve' ? 'reservado' : action === 'release' ? 'liberado' : 'vendido'}`, 'success')
  1029→      onPatched(d.space)
  1030→    } catch {
  1031→      toast('Error de red', 'error')

  1033→      setSaving(false)
  1034→    }
  1035→  }

  1037→  const qrTarget = useMemo(() => {
  1038→    if (!space?.provider?.websiteSlug) return null
  1039→    if (typeof window !== 'undefined') {
  1040→      return `${window.location.origin}/w/${space.provider.websiteSlug}`
  1041→    }
  1042→    return `/w/${space.provider.websiteSlug}`
  1043→  }, [space])
  1044→
  1045→  if (!space) return null
  1046→
  1047→  return (
  1048→    <Dialog open={!!space} onOpenChange={(o) => !o && onClose()}>
  1049→      <DialogContent className="max-w-md">
  1050→        <DialogHeader>
  1051→          <DialogTitle className="flex items-center gap-2">
  1052→            Espacio #{space.position}
  1053→            <Badge className={
  1054→              isAvailable ? 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100'
  1055→              : isReserved ? 'bg-amber-100 text-amber-700 hover:bg-amber-100'
  1056→              : 'bg-rose-100 text-rose-700 hover:bg-rose-100'
  1057→            }>
  1058→              {STATUS_LABELS[space.status] || space.status}
  1059→            </Badge>
  1060→          </DialogTitle>
  1061→          <DialogDescription>
  1062→            {panel?.label} · Fila {row + 1}, Col {col + 1} · 5×5 cm · {fmtUsd(space.price || 20)}
  1063→          </DialogDescription>
  1064→        </DialogHeader>
  1065→
  1066→        <div className="space-y-3">
  1067→          {/* Provider info si ya tiene */}
  1068→          {(isSold || isReserved) && space.provider && (
  1069→            <div className="rounded-lg bg-muted/50 p-3 flex items-center gap-3">
  1070→              {space.provider.logoUrl ? (
  1071→                <img src={space.provider.logoUrl} alt="" className="size-10 rounded object-cover" />
  1072→              ) : (
  1073→                <div className="size-10 rounded bg-muted flex items-center justify-center">
  1074→                  <Store className="size-5 text-muted-foreground" />
  1075→                </div>
  1076→              )}
  1077→              <div className="min-w-0">
  1078→                <p className="font-medium truncate">{space.provider.companyName}</p>
  1079→                <p className="text-xs text-muted-foreground">
  1080→                  {isSold && space.soldAt ? `Vendido el ${fmtDate(space.soldAt)}` : ''}
  1081→                  {isReserved && space.reservedAt ? `Reservado el ${fmtDateTime(space.reservedAt)}` : ''}
  1082→                </p>
  1083→              </div>
  1084→            </div>

  1247→function downloadBlob(content: string, filename: string, mime: string) {
  1248→  const blob = new Blob([content], { type: mime })
  1249→  const url = URL.createObjectURL(blob)
  1250→  const a = document.createElement('a')
  1251→  a.href = url
  1252→  a.download = filename
  1253→  document.body.appendChild(a)
  1254→  a.click()
  1255→  document.body.removeChild(a)
  1256→  setTimeout(() => URL.revokeObjectURL(url), 1000)
  1257→}

  1259→function buildSVG(data: BagPlanData): string {
  1260→  const cellSize = 20
  1261→  const panelGap = 30
  1262→  const labelHeight = 24
  1263→  let totalHeight = 0
  1264→  const maxWidth = Math.max(...BAG_PANELS.map((p) => p.cols)) * cellSize
  1265→
  1266→  const panels = BAG_PANELS.map((p) => {
  1267→    const y = totalHeight + labelHeight
  1268→    const cells: string[] = []
  1269→    for (let pos = p.start; pos <= p.end; pos++) {
  1270→      const idx = pos - p.start
  1271→      const r = Math.floor(idx / p.cols)
  1272→      const c = idx % p.cols
  1273→      const x = c * cellSize
  1274→      const cy = y + r * cellSize
  1275→      const space = data.spaces.find((s) => s.position === pos)
  1276→      let fill = '#f5f5f5'
  1277→      let stroke = '#ddd'
  1278→      if (space?.status === 'AVAILABLE') { fill = '#ecfdf5'; stroke = '#a7f3d0' }
  1279→      else if (space?.status === 'RESERVED') { fill = '#fef3c7'; stroke = '#fcd34d' }
  1280→      else if (space?.status === 'SOLD') {
  1281→        const color = space.provider?.companyName ? sponsorColor(space.provider.companyName) : '#fecdd3'
  1282→        const { h, s, l } = parseHslLocal(color)
  1283→        fill = `hsl(${h}, ${s}%, ${Math.min(l + 25, 88)}%)`
  1284→        stroke = color
  1285→      }
  1286→      cells.push(`<rect x="${x}" y="${cy}" width="${cellSize}" height="${cellSize}" fill="${fill}" stroke="${stroke}" />`)
  1287→      cells.push(`<text x="${x + cellSize / 2}" y="${cy + cellSize / 2 + 3}" font-size="7" text-anchor="middle" fill="#555">${pos}</text>`)
  1288→    }
  1289→    const panelSVG = `
  1290→      <text x="0" y="${totalHeight + 16}" font-size="13" font-weight="bold" fill="#064e3b">${p.label} · ${p.widthCm}×${p.heightCm} cm</text>
  1291→      ${cells.join('\n')}
  1292→    `
  1293→    totalHeight = y + p.rows * cellSize + panelGap
  1294→    return panelSVG
  1295→  })
  1296→
  1297→  return `<svg xmlns="http://www.w3.org/2000/svg" width="${maxWidth}" height="${totalHeight}" viewBox="0 0 ${maxWidth} ${totalHeight}">
  1298→  <rect width="${maxWidth}" height="${totalHeight}" fill="white" />
  1299→  <text x="0" y="14" font-size="14" font-weight="bold" fill="#065f46">${data.edition.name}</text>
  1300→  ${panels.join('\n')}
  1301→</svg>`
  1302→}

  1304→function parseHslLocal(hsl: string): { h: number; s: number; l: number } {
  1305→  const m = hsl.match(/hsl\(\s*(-?\d+(?:\.\d+)?)\s*,\s*(-?\d+(?:\.\d+)?)%\s*,\s*(-?\d+(?:\.\d+)?)%\s*\)/)
  1306→  if (!m) return { h: 150, s: 45, l: 55 }
  1307→  return { h: parseFloat(m[1]), s: parseFloat(m[2]), l: parseFloat(m[3]) }
  1308→}
