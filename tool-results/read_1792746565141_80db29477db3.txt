   795→        /**
   796→         * Recalculate Financial Metrics & Breakdown

   798→        function recalculateMetrics() {
   799→            let availableCount = 0;
   800→            let reservedCount = 0;
   801→            let soldCount = 0;
   802→
   803→            let soldRevenue = 0;
   804→            let reservedRevenue = 0;
   805→            let potentialRevenue = 0;
   806→
   807→            const zoneStats = {};
   808→
   809→            Object.values(slotsData).forEach(s => {
   810→                if (!zoneStats[s.zoneId]) {
   811→                    zoneStats[s.zoneId] = { name: s.zoneName, total: 0, sold: 0, reserved: 0 };
   812→                }
   813→                zoneStats[s.zoneId].total++;
   814→
   815→                const slotVal = parseFloat(s.price) || standardPrice;
   816→
   817→                if (s.status === 'AVAILABLE') {
   818→                    availableCount++;
   819→                    potentialRevenue += slotVal;
   820→                } else if (s.status === 'RESERVED') {
   821→                    reservedCount++;
   822→                    reservedRevenue += slotVal;
   823→                    zoneStats[s.zoneId].reserved++;
   824→                } else if (s.status === 'SOLD') {
   825→                    soldCount++;
   826→                    soldRevenue += slotVal;
   827→                    zoneStats[s.zoneId].sold++;
   828→                }
   829→            });
   830→
   831→            // Update Header Stats
   832→            document.getElementById('hdr-available').textContent = availableCount;
   833→            document.getElementById('hdr-reserved').textContent = reservedCount;
   834→            document.getElementById('hdr-sold').textContent = soldCount;
   835→            document.getElementById('hdr-revenue').textContent = `$${soldRevenue} USD`;
   836→
   837→            // Update Financial Tab Stats
   838→            document.getElementById('fin-total-sold').textContent = `$${soldRevenue} USD`;
   839→            document.getElementById('fin-total-reserved').textContent = `$${reservedRevenue} USD`;
   840→            document.getElementById('fin-total-potential').textContent = `$${potentialRevenue} USD`;
   841→            document.getElementById('fin-grand-total').textContent = `$${soldRevenue + reservedRevenue + potentialRevenue} USD`;
   842→
   843→            // Zone Breakdown Render
   844→            const bdContainer = document.getElementById('fin-zones-breakdown');
   845→            bdContainer.innerHTML = '';
   846→            Object.keys(zoneStats).forEach(zKey => {
   847→                const z = zoneStats[zKey];
   848→                const soldPct = Math.round(((z.sold + z.reserved) / z.total) * 100);
   849→
   850→                const row = document.createElement('div');
   851→                row.className = 'flex flex-col gap-1 border-b border-slate-700/50 pb-1.5';
   852→                row.innerHTML = `
   853→                    <div class="flex justify-between font-semibold">
   854→                        <span class="${zKey === 'BASE' ? 'text-sky-400' : 'text-slate-300'}">${z.name}</span>
   855→                        <span class="text-slate-200">${z.sold + z.reserved}/${z.total} (${soldPct}%)</span>
   856→                    </div>
   857→                    <div class="w-full bg-slate-950 h-1.5 rounded-full overflow-hidden flex">
   858→                        <div class="bg-emerald-500 h-full" style="width: ${(z.sold/z.total)*100}%"></div>
   859→                        <div class="bg-amber-500 h-full" style="width: ${(z.reserved/z.total)*100}%"></div>
   860→                    </div>
   861→                `;
   862→                bdContainer.appendChild(row);
   863→            });
   864→
   865→            renderActiveSponsorList();
   866→        }
