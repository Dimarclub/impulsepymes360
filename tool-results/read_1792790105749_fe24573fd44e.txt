   659→   659→function LivePanel({ panel, spaces, view }: { panel: BagPanel; spaces: BagSpace[]; view: 'current' | 'next' }) {
   660→   660→  const byPos = useMemo(() => {
   661→   661→    const m = new Map<number, BagSpace>()
   662→   662→    for (const s of spaces) m.set(s.position, s)
   663→   663→    return m
   664→   664→  }, [spaces])
   665→   665→
   666→   666→  const positions: number[] = []
   667→   667→  for (let i = panel.start; i <= panel.end; i++) positions.push(i)
   668→   668→
   669→   669→  return (
   670→   670→    <div className="rounded-xl p-4" style={{ background: C.card, border: `1px solid ${C.border}` }}>

   675→   675→        </div>

   677→   677→      </div>

   679→   679→        className="grid"
   680→   680→        style={{ gridTemplateColumns: `repeat(${panel.cols}, minmax(0, 1fr))`, gap: 2 }}
   681→   681→      >
   682→   682→        {positions.map((pos) => {
   683→   683→          const s = byPos.get(pos)
   684→   684→          let bg = C.bg
   685→   685→          let title = `#${pos} · Disponible`
   686→   686→          if (view === 'current') {
   687→   687→            // Todos vendidos → color por sponsor

   689→   689→            if (name) {
   690→   690→              bg = sponsorColor(name)
   691→   691→              title = `#${pos} · ${name}`
   692→   692→            }
   693→   693→          } else {
   694→   694→            // next → color por status
   695→   695→            if (s?.status === 'SOLD') {
   696→   696→              const name = s.buyerName
   697→   697→              bg = name ? sponsorColor(name) : C.rose

   701→   701→              title = `#${pos} · Reservado`
   702→   702→            } else {
   703→   703→              bg = `${C.emerald}33`
   704→   704→              title = `#${pos} · Disponible`
   705→   705→            }
   706→   706→          }
   707→   707→          return (
   708→   708→            <div
   709→   709→              key={pos}
   710→   710→              title={title}
   711→   711→              style={{
   712→   712→                aspectRatio: '1 / 1',
   713→   713→                background: bg,
   714→   714→                borderRadius: 2,
   715→   715→                minWidth: 6,
   716→   716→                cursor: 'default',
   717→   717→              }}
   718→   718→            />
   719→   719→          )
   720→   720→        })}
   721→   721→      </div>
