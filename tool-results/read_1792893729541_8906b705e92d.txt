   495→// ===================== PLANO LIVE =====================
   496→function PlanoLive({ data, loading, onBuy }: { data: BagPlanData | null; loading: boolean; onBuy: () => void }) {
   497→  const [view, setView] = useState<'current' | 'next'>('current')
   498→
   499→  const edition = view === 'current' ? data?.currentEdition : data?.nextEdition
   500→  const spaces = view === 'current' ? data?.currentSpaces : data?.nextSpaces
   501→  const stats = view === 'current' ? data?.currentStats : data?.nextStats
   502→
   503→  // Para la campaña actual (todos vendidos), agrupamos sponsors para la leyenda
   504→  const sponsors = useMemo(() => {
   505→    if (!spaces) return []
   506→    const map = new Map<string, number>()
   507→    for (const s of spaces) {
   508→      if (s.status !== 'SOLD') continue
   509→      const n = s.buyerName || ''
   510→      if (!n) continue
   511→      map.set(n, (map.get(n) || 0) + 1)
   512→    }
   513→    return Array.from(map.entries()).sort((a, b) => b[1] - a[1]).slice(0, 12)
   514→  }, [spaces])
   515→
   516→  return (

   657→function LivePanel({ panel, spaces, view }: { panel: BagPanel; spaces: BagSpace[]; view: 'current' | 'next' }) {
   658→  const byPos = useMemo(() => {
   659→    const m = new Map<number, BagSpace>()
   660→    for (const s of spaces) m.set(s.position, s)
   661→    return m
   662→  }, [spaces])
   663→
   664→  const positions: number[] = []
   665→  for (let i = panel.start; i <= panel.end; i++) positions.push(i)
   666→
   667→  return (
   668→    <div className="rounded-xl p-4" style={{ background: C.card, border: `1px solid ${C.border}` }}>
   669→      <div className="flex items-center justify-between mb-3">
   670→        <div>
   671→          <p style={{ fontSize: 14, fontWeight: 600, color: C.text }}>{panel.label}</p>
   672→          <p style={{ fontSize: 11, color: C.muted }}>{panel.widthCm}×{panel.heightCm} cm · {panel.cols}×{panel.rows}</p>
   673→        </div>
   674→        <span style={{ fontSize: 11, color: C.muted }}>#{panel.start}–{panel.end}</span>
   675→      </div>
   676→      <div
   677→        className="grid"
   678→        style={{ gridTemplateColumns: `repeat(${panel.cols}, minmax(0, 1fr))`, gap: 2 }}
   679→      >
   680→        {positions.map((pos) => {
   681→          const s = byPos.get(pos)
   682→          let bg = C.bg
   683→          let title = `#${pos} · Disponible`
   684→          if (view === 'current') {
   685→            // Todos vendidos → color por sponsor
   686→            const name = s?.buyerName
   687→            if (name) {
   688→              bg = sponsorColor(name)
   689→              title = `#${pos} · ${name}`
   690→            }
   691→          } else {
   692→            // next → color por status
   693→            if (s?.status === 'SOLD') {
   694→              const name = s.buyerName
   695→              bg = name ? sponsorColor(name) : C.rose
   696→              title = `#${pos} · Vendido${name ? ` · ${name}` : ''}`
   697→            } else if (s?.status === 'RESERVED') {
   698→              bg = C.amber
   699→              title = `#${pos} · Reservado`
   700→            } else {
   701→              bg = `${C.emerald}33`
   702→              title = `#${pos} · Disponible`
   703→            }
   704→          }
   705→          return (
   706→            <div
   707→              key={pos}
   708→              title={title}
   709→              style={{
   710→                aspectRatio: '1 / 1',
   711→                background: bg,
   712→                borderRadius: 2,
   713→                minWidth: 6,
   714→                cursor: 'default',
   715→              }}
   716→            />
   717→          )
   718→        })}
   719→      </div>
   720→    </div>
   721→  )
   722→}
