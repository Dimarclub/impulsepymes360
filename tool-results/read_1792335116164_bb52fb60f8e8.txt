   228→// ---------- TreeNode recursivo ----------
   229→function TreeNode({ node, level }: { node: AffiliateNode; level: number }) {
   230→  const hasChildren = node.children.length > 0
   231→  const childCount = node.children.length
   232→
   233→  return (
   234→    <div className="relative flex flex-col items-center">
   235→      <NodeCard node={node} level={level} />
   236→
   237→      {hasChildren && (
   238→        <>
   239→          {/* línea vertical hacia abajo */}
   240→          <div className="w-px h-5 bg-border" aria-hidden />
   241→
   242→          {/* contenedor de hijos */}
   243→          <div
   244→            className="relative flex justify-center"
   245→            style={{ gap: 0 }}
   246→          >
   247→            {/* línea horizontal conectora (solo si hay 2+ hijos) */}
   248→            {childCount > 1 && (
   249→              <div
   250→                className="absolute top-0 h-px bg-border"
   251→                style={{
   252→                  left: `calc(50% / ${childCount})`,
   253→                  right: `calc(50% / ${childCount})`,
   254→                }}
   255→                aria-hidden
   256→              />
   257→            )}
   258→
   259→            {node.children.map((child) => (
   260→              <div
   261→                key={child.id}
   262→                className="relative flex flex-col items-center px-2"
   263→              >
   264→                {/* línea vertical hacia arriba (hijo) */}
   265→                <div className="w-px h-5 bg-border" aria-hidden />
   266→                <TreeNode node={child} level={level + 1} />
   267→              </div>
   268→            ))}
   269→          </div>
   270→        </>
   271→      )}
   272→    </div>
   273→  )
   274→}

   461→// ===================== TAB: COMISIONES =====================
   462→function ComisionesTab({ userId }: { userId?: string }) {
   463→  const [commissions, setCommissions] = useState<Commission[]>([])
   464→  const [loading, setLoading] = useState(true)
   465→  const [filter, setFilter] = useState<string>('ALL')
   466→
   467→  useEffect(() => {
   468→    let mounted = true
   469→    const url = userId
   470→      ? `/api/affiliates/commissions?userId=${encodeURIComponent(userId)}`
   471→      : '/api/affiliates/commissions'
   472→    fetch(url)
   473→      .then((r) => r.json())
   474→      .then((d: Commission[]) => {
   475→        if (mounted) {
   476→          setCommissions(d)
   477→          setLoading(false)
   478→        }
   479→      })
   480→      .catch(() => mounted && setLoading(false))
   481→    return () => {
   482→      mounted = false
   483→    }
   484→  }, [userId])
   485→
   486→  const totalsByLevel = useMemo(() => {
   487→    const map: Record<string, number> = { N0: 0, N1: 0, N2: 0, N3: 0 }
   488→    for (const c of commissions) {
   489→      const key = `N${c.level}`
   490→      map[key] = (map[key] || 0) + c.amount
   491→    }
   492→    return map
   493→  }, [commissions])
   494→
   495→  const total = useMemo(
   496→    () => commissions.reduce((s, c) => s + c.amount, 0),
   497→    [commissions],
   498→  )
   499→
   500→  const filtered = useMemo(() => {
   501→    if (filter === 'ALL') return commissions
   502→    return commissions.filter((c) => `N${c.level}` === filter)
   503→  }, [commissions, filter])
   504→
   505→  if (loading) return <TableSkeleton rows={6} />
   506→
   507→  return (
   508→    <div className="space-y-6">
   509→      {/* Resumen */}
   510→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   511→        <MiniStat
   512→          icon={<Wallet className="size-4" />}
   513→          label="Total comisiones"
   514→          value={fmtUSD(total)}
   515→          accent="primary"
   516→        />

   518→          icon={<Coins className="size-4" />}
   519→          label="Nivel 0 (3%)"
   520→          value={fmtUSD(totalsByLevel.N0 || 0)}
   521→          accent="primary"
   522→        />
   523→        <MiniStat
   524→          icon={<Coins className="size-4" />}
   525→          label="Nivel 1 (2%)"
   526→          value={fmtUSD(totalsByLevel.N1 || 0)}
   527→          accent="accent"
   528→        />
   529→        <MiniStat
   530→          icon={<Coins className="size-4" />}
   531→          label="Niveles 2+3"
   532→          value={fmtUSD((totalsByLevel.N2 || 0) + (totalsByLevel.N3 || 0))}
   533→          accent="teal"
   534→        />
   535→      </div>
   536→
   537→      {/* Filtro + tabla */}
   538→      <Card className="p-0">
   539→        <CardHeader>
   540→          <div className="flex items-center justify-between gap-3 flex-wrap">
   541→            <div>
   542→              <CardTitle>Historial de comisiones</CardTitle>
   543→              <CardDescription>
   544→                {filtered.length} registros · Total {fmtUSD(filtered.reduce((s, c) => s + c.amount, 0))}
   545→              </CardDescription>
   546→            </div>
   547→            <div className="flex items-center gap-2">
   548→              <Filter className="size-4 text-muted-foreground" />

   559→                </SelectContent>
   560→              </Select>
   561→            </div>
   562→          </div>
   563→        </CardHeader>

   631→function MisVentasTab({ userId }: { userId?: string }) {
   632→  const [orders, setOrders] = useState<Order[]>([])
   633→  const [loading, setLoading] = useState(true)
   634→
   635→  useEffect(() => {
   636→    let mounted = true
   637→    const url = userId
   638→      ? `/api/affiliates/sales?userId=${encodeURIComponent(userId)}`
   639→      : '/api/affiliates/sales'
   640→    fetch(url)
   641→      .then((r) => r.json())
   642→      .then((d: Order[]) => {
   643→        if (mounted) {
   644→          setOrders(d)
   645→          setLoading(false)
   646→        }
   647→      })
   648→      .catch(() => mounted && setLoading(false))
   649→    return () => {
   650→      mounted = false
   651→    }
   652→  }, [userId])
   653→
   654→  if (loading) return <TableSkeleton rows={6} />
   655→
   656→  const totalVentas = orders.reduce((s, o) => s + o.total, 0)
   657→  const totalComisionN0 = orders.reduce((s, o) => s + o.total * N0_RATE, 0)
   658→
   659→  return (
   660→    <div className="space-y-6">
   661→      {/* Resumen */}
   662→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   663→        <MiniStat
   664→          icon={<ShoppingBag className="size-4" />}
   665→          label="Total ventas"
   666→          value={fmtNum(orders.length)}
   667→          accent="primary"
   668→        />
   669→        <MiniStat
   670→          icon={<TrendingUp className="size-4" />}
   671→          label="Monto vendido"
   672→          value={fmtUSD(totalVentas)}
   673→          accent="primary"
   674→        />
   675→        <MiniStat
   676→          icon={<Coins className="size-4" />}
   677→          label="Comisión N0 (3%)"
   678→          value={fmtUSD(totalComisionN0)}
   679→          accent="accent"
   680→        />
   681→        <MiniStat
   682→          icon={<UserCircle2 className="size-4" />}
   683→          label="Ticket promedio"
   684→          value={orders.length > 0 ? fmtUSD(totalVentas / orders.length) : '$0.00'}
   685→          accent="teal"
   686→        />
   687→      </div>
   688→
   689→      {/* Tabla */}
   690→      <Card className="p-0">
   691→        <CardHeader>
   692→          <CardTitle>Tus ventas directas</CardTitle>
   693→          <CardDescription>
   694→            Generas {N0_RATE * 100}% de comisión por cada venta propia
   695→          </CardDescription>
   696→        </CardHeader>
   697→        <CardContent>
   698→          <div className="max-h-[480px] overflow-y-auto scroll-thin rounded-md border border-border/60">
   699→            <Table>
   700→              <TableHeader className="sticky top-0 bg-card z-10">
   701→                <TableRow>
   702→                  <TableHead className="px-4">Pedido</TableHead>
   703→                  <TableHead className="px-4">Fecha</TableHead>
   704→                  <TableHead className="px-4 text-center">Items</TableHead>
   705→                  <TableHead className="px-4 text-right">Total</TableHead>
   706→                  <TableHead className="px-4">Estado</TableHead>
   707→                  <TableHead className="px-4 text-right">Comisión N0</TableHead>
   708→                </TableRow>
   709→              </TableHeader>
   710→              <TableBody>
   711→                {orders.map((o) => (
   712→                  <TableRow key={o.id}>
   713→                    <TableCell className="px-4 font-mono text-xs text-foreground">
   714→                      {o.orderNumber}

   716→                    <TableCell className="px-4 text-xs text-muted-foreground">

   720→                      {o.items?.length || 0}

   723→                      {fmtUSD(o.total)}
   724→                    </TableCell>
   725→                    <TableCell className="px-4">
   726→                      <Badge
   727→                        variant="outline"
   728→                        className={cn(
   729→                          'border',
   730→                          ORDER_STATUS_COLORS[o.status] ||
   731→                            'bg-muted text-muted-foreground border-border',
   732→                        )}
   733→                      >
   734→                        {ORDER_STATUS_LABELS[o.status] || o.status}
   735→                      </Badge>
   736→                    </TableCell>
   737→                    <TableCell className="px-4 text-right font-semibold text-primary">
   738→                      {fmtUSD(o.total * N0_RATE)}
   739→                    </TableCell>
   740→                  </TableRow>
   741→                ))}
   742→                {orders.length === 0 && (
   743→                  <TableRow>
   744→                    <TableCell colSpan={6} className="text-center text-muted-foreground py-8">
