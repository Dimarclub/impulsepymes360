   102→function AffiliatesTab() {
   103→  const toast = useAppStore((s) => s.toast)
   104→  const currentUser = useAppStore((s) => s.currentUser)
   105→  const [tree, setTree] = useState<any>(null)
   106→  const [loading, setLoading] = useState(true)
   107→  const [registerOpen, setRegisterOpen] = useState(false)
   108→  const [regForm, setRegForm] = useState({ name: '', email: '', phone: '', city: '', password: '' })
   109→
   110→  const fetchTree = useCallback(async () => {
   111→    setLoading(true)
   112→    try {
   113→      const res = await fetch(`/api/affiliates/tree?userId=${currentUser?.id}`)
   114→      if (!res.ok) { setTree(null); return }
   115→      setTree(await res.json())
   116→    } catch { setTree(null) }
   117→    finally { setLoading(false) }
   118→  }, [currentUser?.id])
   119→
   120→  useEffect(() => { fetchTree() }, [fetchTree])
   121→
   122→  const registerAffiliate = async () => {
   123→    if (!regForm.name || !regForm.email || !regForm.password) { toast('Completa los campos', 'error'); return }
   124→    try {
   125→      const res = await fetch('/api/auth/register', {
   126→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   127→        body: JSON.stringify({ ...regForm, role: 'AFFILIATE', sponsorCode: currentUser?.affiliateCode }),
   128→      })
   129→      const d = await res.json()
   130→      if (!res.ok) { toast(d.error || 'Error', 'error'); return }
   131→      toast(`¡Afiliado registrado: ${d.name}!`, 'success')
   132→      setRegisterOpen(false)
   133→      setRegForm({ name: '', email: '', phone: '', city: '', password: '' })
   134→      fetchTree()
   135→    } catch { toast('Error', 'error') }
   136→  }
   137→
   138→  if (loading) return <Skeleton className="h-64" />
   139→  if (!tree) return <Card><CardContent className="py-8 text-center text-muted-foreground">No se pudo cargar la red.</CardContent></Card>
   140→
   141→  return (
   142→    <div className="space-y-4">
   143→      <div className="flex items-center justify-between">
   144→        <div>
   145→          <h2 className="text-lg font-bold">Mi red de afiliados</h2>
   146→          <p className="text-sm text-muted-foreground">Registra nuevos afiliados y gestiona tu red de 3 niveles</p>
   147→        </div>
   148→        <Button onClick={() => setRegisterOpen(true)}><Plus className="size-4 mr-1" /> Nuevo afiliado</Button>
   149→      </div>
   150→
   151→      {/* Stats */}
   152→      <div className="grid grid-cols-3 gap-3">
   153→        <Card><CardContent className="p-4 text-center"><p className="text-2xl font-bold text-primary">{tree.stats?.referrals || 0}</p><p className="text-xs text-muted-foreground">Referidos directos</p></CardContent></Card>
   154→        <Card><CardContent className="p-4 text-center"><p className="text-2xl font-bold text-amber-600">{fmtUsd(tree.stats?.commissions || 0)}</p><p className="text-xs text-muted-foreground">Comisiones totales</p></CardContent></Card>
   155→        <Card><CardContent className="p-4 text-center"><p className="text-2xl font-bold">{tree.stats?.sales || 0}</p><p className="text-xs text-muted-foreground">Ventas</p></CardContent></Card>
   156→      </div>

   158→      {/* Árbol simplificado */}
   159→      <Card>
   160→        <CardHeader><CardTitle className="text-base">Mi red (N1 → N2 → N3)</CardTitle></CardHeader>
   161→        <CardContent>
   162→          {tree.children?.length === 0 ? (
   163→            <p className="text-center text-sm text-muted-foreground py-8">No tienes afiliados registrados todavía. ¡Registra tu primer afiliado!</p>

   289→// ============ TAREAS ============
   290→function TasksTab() {
   291→  const toast = useAppStore((s) => s.toast)
   292→  const [progress, setProgress] = useState<any>(null)
   293→  const [loading, setLoading] = useState(true)
   294→
   295→  const fetchProgress = useCallback(async () => {
   296→    setLoading(true)
   297→    try {
   298→      const res = await fetch('/api/tasks/progress')
   299→      if (!res.ok) { setProgress(null); return }
   300→      setProgress(await res.json())
   301→    } catch { setProgress(null) }
   302→    finally { setLoading(false) }
   303→  }, [])
   304→
   305→  useEffect(() => { fetchProgress() }, [fetchProgress])
   306→
   307→  const toggleTask = async (task: string) => {
   308→    const res = await fetch('/api/tasks/progress', {
   309→      method: 'PATCH', headers: { 'Content-Type': 'application/json' },
   310→      body: JSON.stringify({ [task]: !progress[task] }),
   311→    })
   312→    if (res.ok) setProgress(await res.json())
   313→  }
   314→
   315→  const claimReward = async () => {
   316→    const res = await fetch('/api/tasks/complete', { method: 'POST' })
   317→    const d = await res.json()
   318→    if (!res.ok) { toast(d.error || 'Error', 'error'); return }
   319→    toast('¡Recompensa reclamada!', 'success')
   320→    fetchProgress()
   321→  }
   322→
   323→  if (loading || !progress) return <Skeleton className="h-64" />
   324→
   325→  const tasks = [
   326→    { key: 'scanDone', label: 'Escanear código QR', icon: QrCode, desc: 'Escanea el QR del bolso o grid' },
   327→    { key: 'identifyDone', label: 'Identificar empresa', icon: Store, desc: 'Identifica qué empresa es el QR' },
   328→    { key: 'visitWebDone', label: 'Visitar sitio web', icon: ArrowRight, desc: 'Navega el sitio autogenerado' },
   329→    { key: 'shareDone', label: 'Compartir oferta', icon: Share2, desc: 'Comparte en WhatsApp/redes' },
   330→    { key: 'commentDone', label: 'Dejar comentario', icon: MessageSquare, desc: 'Comenta en el sitio' },
   331→    { key: 'photoDone', label: 'Foto en tienda', icon: Camera, desc: 'Visita la tienda y toma foto' },
   332→  ]
   333→  const completedCount = tasks.filter((t) => progress[t.key]).length
   334→  const progressPercent = (completedCount / tasks.length) * 100
   335→
   336→  return (
   337→    <div className="space-y-4">
   338→      <Card>
   339→        <CardHeader>
   340→          <CardTitle className="text-base flex items-center gap-2"><CheckCircle2 className="size-5 text-primary" /> Tareas del bolso</CardTitle>
   341→          <CardDescription>Completa las 6 tareas para recibir bolso + álbum + sticker</CardDescription>
   342→        </CardHeader>
   343→        <CardContent className="space-y-3">
   344→          <div>
   345→            <div className="flex justify-between text-sm mb-1">
   346→              <span>Progreso: {completedCount}/{tasks.length}</span>
   347→              <span className="font-bold text-primary">{progressPercent}%</span>
   348→            </div>
   349→            <Progress value={progressPercent} className="h-2" />
   350→          </div>
   351→          {tasks.map((t) => {
   352→            const done = progress[t.key]; const Icon = t.icon
   353→            return (
   354→              <button key={t.key} onClick={() => toggleTask(t.key)}
   355→                className={cn('w-full flex items-center gap-3 p-3 rounded-lg border transition-all text-left',
   356→                  done ? 'bg-emerald-50 border-emerald-300' : 'bg-card border-border hover:bg-muted/50')}>
   357→                <div className={cn('size-9 rounded-full flex items-center justify-center shrink-0', done ? 'bg-emerald-500' : 'bg-muted')}>
   358→                  {done ? <CheckCircle2 className="size-5 text-white" /> : <Icon className="size-5 text-muted-foreground" />}
   359→                </div>
   360→                <div className="flex-1">
   361→                  <p className={cn('text-sm font-medium', done && 'line-through text-muted-foreground')}>{t.label}</p>
   362→                  <p className="text-xs text-muted-foreground">{t.desc}</p>
   363→                </div>
   364→              </button>
   365→            )
   366→          })}
   367→          {progress.completed && !progress.rewardClaimed && (
   368→            <Button className="w-full" size="lg" onClick={claimReward}><Gift className="size-5 mr-2" />Reclamar recompensa</Button>
   369→          )}
   370→          {progress.rewardClaimed && (
   371→            <div className="rounded-lg bg-emerald-50 border border-emerald-200 p-4 text-center">
   372→              <CheckCircle2 className="size-8 mx-auto text-emerald-600 mb-2" />
   373→              <p className="font-medium text-emerald-800">¡Recompensa reclamada!</p>
   374→            </div>
   375→          )}
   376→        </CardContent>
   377→      </Card>
   378→    </div>
   379→  )
   380→}

   408→function LogisticsTab() {
   409→  const toast = useAppStore((s) => s.toast)
   410→  const setModule = useAppStore((s) => s.setModule)
   411→  const [form, setForm] = useState({
   412→    taskType: 'SOBRE' as 'SOBRE' | 'CARGA',
   413→    pickupAddress: '',
   414→    deliveryAddress: '',
   415→    recipientName: '',
   416→    recipientPhone: '',
   417→    notes: '',
   418→  })
   419→  const [submitting, setSubmitting] = useState(false)
   420→  const [deliveries, setDeliveries] = useState<any[]>([])
   421→  const [loadingList, setLoadingList] = useState(true)
   422→
   423→  const fetchDeliveries = useCallback(async () => {
   424→    setLoadingList(true)
   425→    try {
   426→      const res = await fetch('/api/deliveries/request')
   427→      const d = await res.json()
   428→      if (res.ok && d?.deliveries) setDeliveries(d.deliveries)
   429→      else setDeliveries([])
   430→    } catch {
   431→      setDeliveries([])
   432→    } finally {
   433→      setLoadingList(false)
   434→    }
   435→  }, [])
   436→
   437→  useEffect(() => { fetchDeliveries() }, [fetchDeliveries])
   438→
   439→  const submit = async () => {
   440→    if (!form.pickupAddress.trim() || !form.deliveryAddress.trim()) {
   441→      toast('Completa las direcciones de recogida y entrega', 'error')
   442→      return
   443→    }
   444→    setSubmitting(true)
   445→    try {
   446→      const res = await fetch('/api/deliveries/request', {
   447→        method: 'POST',
   448→        headers: { 'Content-Type': 'application/json' },
   449→        body: JSON.stringify({
   450→          taskType: form.taskType,
   451→          pickupAddress: form.pickupAddress.trim(),
   452→          deliveryAddress: form.deliveryAddress.trim(),
   453→          recipientName: form.recipientName.trim() || undefined,
   454→          recipientPhone: form.recipientPhone.trim() || undefined,
   455→          notes: form.notes.trim() || undefined,
   456→        }),
   457→      })
   458→      const d = await res.json()
   459→      if (!res.ok) {
   460→        toast(d?.error || 'Error al enviar la solicitud', 'error')
   461→        return
   462→      }
   463→      toast('Solicitud enviada. El Delta Master asignará un mensajero pronto.', 'success')
   464→      setForm({
   465→        taskType: 'SOBRE',
   466→        pickupAddress: '',
   467→        deliveryAddress: '',
   468→        recipientName: '',
   469→        recipientPhone: '',
   470→        notes: '',
   471→      })
   472→      fetchDeliveries()
   473→    } catch {
   474→      toast('Error de red al enviar la solicitud', 'error')
   475→    } finally {
   476→      setSubmitting(false)
   477→    }
   478→  }
   479→
   480→  return (
   481→    <div className="space-y-4">
   482→      {/* Banner informativo: solo el Delta Master asigna mensajeros */}
   483→      <div className="rounded-lg border border-amber-200 bg-amber-50 dark:border-amber-900/40 dark:bg-amber-950/20 p-3 flex items-start gap-3">
   484→        <ClipboardList className="size-5 text-amber-600 shrink-0 mt-0.5" />
   485→        <div className="text-sm">
   486→          <p className="font-medium text-amber-900 dark:text-amber-200">¿Cómo funciona el servicio de logística?</p>
   487→          <p className="text-amber-700 dark:text-amber-300 mt-0.5">
   488→            Tú solicitas el envío y el <strong>Delta Master</strong> (administrador) asigna un mensajero.
   489→            Recibirás el nombre del mensajero en cuanto sea asignado.
   490→          </p>
   491→        </div>
   492→      </div>
   493→
   494→      <div className="grid lg:grid-cols-2 gap-4">
   495→        {/* Formulario */}
   496→        <Card>
   497→          <CardHeader>
   498→            <CardTitle className="text-base flex items-center gap-2">
   499→              <Truck className="size-5 text-primary" /> Solicitar servicio de logística
   500→            </CardTitle>
   501→            <CardDescription>
   502→              ¿Necesitas enviar un sobre o carga? Solicita el servicio y nuestro Delta Master asignará un mensajero.
   503→            </CardDescription>
   504→          </CardHeader>
   505→          <CardContent className="space-y-4">
   506→            {/* Tipo de envío */}
   507→            <div className="space-y-1.5">
   508→              <Label>Tipo de envío <span className="text-destructive">*</span></Label>
   509→              <Select value={form.taskType} onValueChange={(v) => setForm({ ...form, taskType: v as 'SOBRE' | 'CARGA' })}>
   510→                <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   511→                <SelectContent>
   512→                  <SelectItem value="SOBRE">Sobre (documentos, sobres pequeños)</SelectItem>
   513→                  <SelectItem value="CARGA">Carga (paquetes, bultos)</SelectItem>
   514→                </SelectContent>
   515→              </Select>
   516→            </div>
   517→
   518→            {/* Dirección de recogida */}
   519→            <div className="space-y-1.5">
   520→              <Label className="flex items-center gap-1.5"><MapPin className="size-3.5 text-muted-foreground" /> Dirección de recogida <span className="text-destructive">*</span></Label>
   521→              <Input
   522→                value={form.pickupAddress}
   523→                onChange={(e) => setForm({ ...form, pickupAddress: e.target.value })}
   524→                placeholder="Ej. Av. Amazonas N34-451, Quito"
   525→              />
   526→            </div>
   527→
   528→            {/* Dirección de entrega */}
   529→            <div className="space-y-1.5">
   530→              <Label className="flex items-center gap-1.5"><MapPin className="size-3.5 text-muted-foreground" /> Dirección de entrega <span className="text-destructive">*</span></Label>
   531→              <Input
   532→                value={form.deliveryAddress}
   533→                onChange={(e) => setForm({ ...form, deliveryAddress: e.target.value })}
   534→                placeholder="Ej. Calle Bolívar 12-34, Guayaquil"
   535→              />
   536→            </div>
   537→
   538→            {/* Destinatario */}
   539→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   540→              <div className="space-y-1.5">
   541→                <Label className="flex items-center gap-1.5"><User className="size-3.5 text-muted-foreground" /> Nombre del destinatario</Label>
   542→                <Input
   543→                  value={form.recipientName}
   544→                  onChange={(e) => setForm({ ...form, recipientName: e.target.value })}
   545→                  placeholder="Opcional"
   546→                />
   547→              </div>
   548→              <div className="space-y-1.5">
   549→                <Label className="flex items-center gap-1.5"><Phone className="size-3.5 text-muted-foreground" /> Teléfono del destinatario</Label>
   550→                <Input
   551→                  value={form.recipientPhone}
   552→                  onChange={(e) => setForm({ ...form, recipientPhone: e.target.value })}
   553→                  placeholder="Opcional"
   554→                />
   555→              </div>
   556→            </div>
   557→
   558→            {/* Notas */}
   559→            <div className="space-y-1.5">
   560→              <Label>Notas</Label>
   561→              <Textarea
   562→                value={form.notes}
   563→                onChange={(e) => setForm({ ...form, notes: e.target.value })}
   564→                placeholder="Descripción del contenido, instrucciones especiales..."
   565→                rows={3}
   566→              />
   567→            </div>
   568→
   569→            <Button
   570→              className="w-full bg-emerald-600 hover:bg-emerald-700 text-white"
   571→              onClick={submit}
   572→              disabled={submitting}
   573→            >
   574→              {submitting ? (
   575→                <><Loader2 className="size-4 mr-1.5 animate-spin" /> Enviando...</>
   576→              ) : (
   577→                <><Send className="size-4 mr-1.5" /> Enviar solicitud</>
   578→              )}
   579→            </Button>
   580→          </CardContent>
   581→        </Card>
   582→
   583→        {/* Tus solicitudes recientes */}
   584→        <Card>
   585→          <CardHeader>
   586→            <CardTitle className="text-base flex items-center gap-2">
   587→              <ClipboardList className="size-5 text-primary" /> Tus solicitudes recientes
   588→            </CardTitle>
   589→            <CardDescription>Solicitudes de logística que has creado</CardDescription>
   590→          </CardHeader>
   591→          <CardContent>
   592→            {loadingList ? (
   593→              <div className="space-y-2">
   594→                <Skeleton className="h-16 w-full" />
   595→                <Skeleton className="h-16 w-full" />
   596→                <Skeleton className="h-16 w-full" />
   597→              </div>
   598→            ) : deliveries.length === 0 ? (
   599→              <div className="text-center py-10 text-sm text-muted-foreground">
   600→                <Truck className="size-10 mx-auto mb-2 opacity-30" />
   601→                No tienes solicitudes de logística todavía.
   602→                <br />Crea la primera con el formulario de la izquierda.
   603→              </div>
   604→            ) : (
   605→              <div className="space-y-2 max-h-[28rem] overflow-y-auto pr-1">
   606→                {deliveries.map((d) => {
   607→                  const st = d.status || 'PENDING'
   608→                  const stLabel = LOGISTICS_STATUS_LABELS[st] || st
   609→                  const stBadge = LOGISTICS_STATUS_BADGES[st] || 'bg-muted text-muted-foreground border-border'
   610→                  const courierName = d.courier?.user?.name
   611→                  return (
   612→                    <div key={d.id} className="rounded-lg border p-3 space-y-2">
   613→                      <div className="flex items-center justify-between gap-2">
   614→                        <div className="flex items-center gap-2 min-w-0">
   615→                          <Badge variant="outline" className="shrink-0">
   616→                            {TASK_TYPE_LABEL[d.taskType] || d.taskType}
   617→                          </Badge>
   618→                          <span className="text-xs text-muted-foreground font-mono truncate">
   619→                            {d.order?.orderNumber || '—'}
   620→                          </span>
   621→                        </div>
   622→                        <Badge variant="outline" className={cn('shrink-0', stBadge)}>{stLabel}</Badge>
   623→                      </div>
   624→                      <div className="text-sm">
   625→                        <p className="flex items-start gap-1.5">
   626→                          <MapPin className="size-3.5 text-muted-foreground shrink-0 mt-0.5" />
   627→                          <span className="truncate">{d.deliveryAddress || '—'}</span>
   628→                        </p>
   629→                        {courierName && (
   630→                          <p className="flex items-center gap-1.5 mt-1 text-emerald-700 dark:text-emerald-300">
   631→                            <User className="size-3.5 shrink-0" /> Mensajero: <strong>{courierName}</strong>
   632→                          </p>
   633→                        )}
   634→                      </div>
   635→                      <p className="text-xs text-muted-foreground">
   636→                        {d.createdAt ? new Date(d.createdAt).toLocaleString('es-EC', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' }) : ''}
   637→                      </p>
   638→                    </div>
   639→                  )
   640→                })}
   641→              </div>
   642→            )}
   643→            <Button variant="ghost" size="sm" className="w-full mt-3" onClick={() => setModule('logistics')}>
   644→              <Truck className="size-4 mr-1.5" /> Ver tablero de logística
   645→            </Button>
   646→          </CardContent>
   647→        </Card>
   648→      </div>
   649→    </div>
   650→  )
   651→}
