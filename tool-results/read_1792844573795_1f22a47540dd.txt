   655→   655→function LivePanel({ panel, spaces, view }: { panel: BagPanel; spaces: BagSpace[]; view: 'current' | 'next' }) {
   656→   656→  const byPos = useMemo(() => {
   657→   657→    const m = new Map<number, BagSpace>()
   658→   658→    for (const s of spaces) m.set(s.position, s)
   659→   659→    return m
   660→   660→  }, [spaces])
   661→   661→
   662→   662→  const positions: number[] = []
   663→   663→  for (let i = panel.start; i <= panel.end; i++) positions.push(i)
   664→   664→
   665→   665→  return (

   671→   671→        </div>

   673→   673→      </div>

   675→   675→        className="grid"
   676→   676→        style={{ gridTemplateColumns: `repeat(${panel.cols}, minmax(0, 1fr))`, gap: 2 }}

   679→   679→          const s = byPos.get(pos)
   680→   680→          let bg = C.bg
   681→   681→          let title = `#${pos} · Disponible`
   682→   682→          if (view === 'current') {
   683→   683→            // Todos vendidos → color por sponsor

   685→   685→            if (name) {

   687→   687→              title = `#${pos} · ${name}`

   691→   691→            if (s?.status === 'SOLD') {
   692→   692→              const name = s.buyerName
   693→   693→              bg = name ? sponsorColor(name) : C.rose

   699→   699→              bg = `${C.emerald}33`
   700→   700→              title = `#${pos} · Disponible`
   701→   701→            }

   703→   703→          return (

   705→   705→              key={pos}
   706→   706→              title={title}
   707→   707→              style={{
   708→   708→                aspectRatio: '1 / 1',
   709→   709→                background: bg,
   710→   710→                borderRadius: 2,
   711→   711→                minWidth: 6,
   712→   712→                cursor: 'default',
   713→   713→              }}

   715→   715→          )
