   497→// ===================== PLANO LIVE =====================
   498→function PlanoLive({ data, loading, onBuy }: { data: BagPlanData | null; loading: boolean; onBuy: () => void }) {
   499→  const [view, setView] = useState<'current' | 'next'>('current')
   500→
   501→  const edition = view === 'current' ? data?.currentEdition : data?.nextEdition
   502→  const spaces = view === 'current' ? data?.currentSpaces : data?.nextSpaces
   503→  const stats = view === 'current' ? data?.currentStats : data?.nextStats
   504→
   505→  // Para la campaña actual (todos vendidos), agrupamos sponsors para la leyenda
   506→  const sponsors = useMemo(() => {
   507→    if (!spaces) return []
   508→    const map = new Map<string, number>()
   509→    for (const s of spaces) {
   510→      if (s.status !== 'SOLD') continue
   511→      const n = s.buyerName || ''
   512→      if (!n) continue

   659→function LivePanel({ panel, spaces, view }: { panel: BagPanel; spaces: BagSpace[]; view: 'current' | 'next' }) {
   660→  const byPos = useMemo(() => {
   661→    const m = new Map<number, BagSpace>()
   662→    for (const s of spaces) m.set(s.position, s)
   663→    return m
   664→  }, [spaces])
   665→
   666→  const positions: number[] = []
   667→  for (let i = panel.start; i <= panel.end; i++) positions.push(i)
   668→
   669→  return (
   670→    <div className="rounded-xl p-4" style={{ background: C.card, border: `1px solid ${C.border}` }}>
   671→      <div className="flex items-center justify-between mb-3">
   672→        <div>
   673→          <p style={{ fontSize: 14, fontWeight: 600, color: C.text }}>{panel.label}</p>
   674→          <p style={{ fontSize: 11, color: C.muted }}>{panel.widthCm}×{panel.heightCm} cm · {panel.cols}×{panel.rows}</p>
   675→        </div>
   676→        <span style={{ fontSize: 11, color: C.muted }}>#{panel.start}–{panel.end}</span>
   677→      </div>
   678→      <div
   679→        className="grid"
   680→        style={{ gridTemplateColumns: `repeat(${panel.cols}, minmax(0, 1fr))`, gap: 2 }}
   681→      >
   682→        {positions.map((pos) => {
   683→          const s = byPos.get(pos)
   684→          let bg = C.bg
   685→          let title = `#${pos} · Disponible`
   686→          if (view === 'current') {
   687→            // Todos vendidos → color por sponsor
   688→            const name = s?.buyerName
   689→            if (name) {
   690→              bg = sponsorColor(name)
   691→              title = `#${pos} · ${name}`
   692→            }
   693→          } else {
   694→            // next → color por status
   695→            if (s?.status === 'SOLD') {
   696→              const name = s.buyerName
   697→              bg = name ? sponsorColor(name) : C.rose
   698→              title = `#${pos} · Vendido${name ? ` · ${name}` : ''}`
   699→            } else if (s?.status === 'RESERVED') {
   700→              bg = C.amber
   701→              title = `#${pos} · Reservado`
   702→            } else {
   703→              bg = `${C.emerald}33`
   704→              title = `#${pos} · Disponible`
   705→            }
   706→          }
   707→          return (
   708→            <div
   709→              key={pos}
   710→              title={title}
   711→              style={{
   712→                aspectRatio: '1 / 1',
   713→                background: bg,
   714→                borderRadius: 2,
   715→                minWidth: 6,
   716→                cursor: 'default',
   717→              }}
   718→            />
   719→          )
   720→        })}
   721→      </div>
   722→    </div>
   723→  )
   724→}
