   465→   465→function CampaignCard({
   466→   466→  campaign,
   467→   467→  onSend,
   468→   468→  onDelete,
   469→   469→}: {
   470→   470→  campaign: EmailCampaign
   471→   471→  onSend: () => void
   472→   472→  onDelete: () => void
   473→   473→}) {
   474→   474→  const status = campaign.status
   475→   475→  const openRate =
   476→   476→    campaign.recipientCount > 0 ? (campaign.openCount / campaign.recipientCount) * 100 : 0
   477→   477→  const clickRate =
   478→   478→    campaign.recipientCount > 0 ? (campaign.clickCount / campaign.recipientCount) * 100 : 0
   479→   479→  const isSent = status === 'SENT'
   480→   480→  const isDraft = status === 'DRAFT'
   481→   481→
   482→   482→  return (
   483→   483→    <Card className="gap-0 overflow-hidden">
   484→   484→      <CardHeader className="pb-3">
   485→   485→        <div className="flex items-start justify-between gap-2">
   486→   486→          <div className="min-w-0">
   487→   487→            <CardTitle className="text-sm truncate">{campaign.name}</CardTitle>
