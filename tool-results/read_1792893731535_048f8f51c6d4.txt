   657→   657→function LivePanel({ panel, spaces, view }: { panel: BagPanel; spaces: BagSpace[]; view: 'current' | 'next' }) {

   659→   659→    const m = new Map<number, BagSpace>()
   660→   660→    for (const s of spaces) m.set(s.position, s)
   661→   661→    return m
   662→   662→  }, [spaces])
   663→   663→
   664→   664→  const positions: number[] = []
   665→   665→  for (let i = panel.start; i <= panel.end; i++) positions.push(i)
   666→   666→
   667→   667→  return (

   673→   673→        </div>

   681→   681→          const s = byPos.get(pos)
   682→   682→          let bg = C.bg
   683→   683→          let title = `#${pos} · Disponible`
   684→   684→          if (view === 'current') {
   685→   685→            // Todos vendidos → color por sponsor

   694→   694→              const name = s.buyerName

   701→   701→              bg = `${C.emerald}33`
   702→   702→              title = `#${pos} · Disponible`
   703→   703→            }
   704→   704→          }
   705→   705→          return (
   706→   706→            <div
   707→   707→              key={pos}
   708→   708→              title={title}
   709→   709→              style={{
   710→   710→                aspectRatio: '1 / 1',
   711→   711→                background: bg,
   712→   712→                borderRadius: 2,
   713→   713→                minWidth: 6,
   714→   714→                cursor: 'default',
   715→   715→              }}
   716→   716→            />
   717→   717→          )
   718→   718→        })}
