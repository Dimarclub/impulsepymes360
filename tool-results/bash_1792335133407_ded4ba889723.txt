    39→`

    41→function Stars({ value }: { value: number }) {
    42→  const full = Math.floor(value)
    43→  const half = value - full >= 0.5
    44→  return (
    45→    <div className="flex items-center gap-0.5" aria-label={`Rating ${value} de 5`}>
    46→      {[0, 1, 2, 3, 4].map((i) => (
    47→        <Star
    48→          key={i}
    49→          className={cn(
    50→            'size-3.5',
    51→            i < full ? 'fill-amber-400 text-amber-400' : i === full && half ? 'fill-amber-200 text-amber-400' : 'text-muted-foreground/40'
    52→          )}
    53→        />
    54→      ))}
    55→      <span className="ml-1 text-xs text-muted-foreground">{value.toFixed(1)}</span>
    56→    </div>
    57→  )
    58→}

    64→export function SuppliersModule() {
    65→  const toast = useAppStore((s) => s.toast)
    66→  const [suppliers, setSuppliers] = useState<SupplierWithRelations[]>([])
    67→  const [categories, setCategories] = useState<Category[]>([])
    68→  const [loading, setLoading] = useState(true)
    69→  const [selectedId, setSelectedId] = useState<string | null>(null)
    70→  const [search, setSearch] = useState('')
    71→
    72→  const [products, setProducts] = useState<Product[]>([])
    73→  const [loadingProducts, setLoadingProducts] = useState(false)
    74→
    75→  // Dialogs
    76→  const [newSupplierOpen, setNewSupplierOpen] = useState(false)
    77→  const [newProductOpen, setNewProductOpen] = useState(false)
    78→  const [bulkOpen, setBulkOpen] = useState(false)
    79→  const [submitting, setSubmitting] = useState(false)
    80→
    81→  // Forms
    82→  const [supplierForm, setSupplierForm] = useState({
    83→    companyName: '', email: '', phone: '', taxId: '', address: '', city: '', category: '',
    84→  })
    85→  const [productForm, setProductForm] = useState({
    86→    name: '', price: '', compareAtPrice: '', stock: '', sku: '', categoryId: '', description: '', imageUrl: '',
    87→  })
    88→  const [csvText, setCsvText] = useState('')
    89→  const fileRef = useRef<HTMLInputElement>(null)
    90→
    91→  const fetchSuppliers = async () => {
    92→    setLoading(true)
    93→    try {
    94→      const res = await fetch('/api/suppliers')
    95→      const data = await res.json()
    96→      setSuppliers(data)
    97→      if (data.length && !selectedId) setSelectedId(data[0].id)
    98→    } catch {
    99→      toast('Error al cargar proveedores', 'error')
   100→    } finally {
   101→      setLoading(false)
   102→    }
   103→  }
   104→
   105→  const fetchCategories = async () => {
   106→    try {
   107→      const res = await fetch('/api/categories')
   108→      const data = await res.json()
   109→      setCategories(data)
   110→    } catch {
   111→      /* noop */
   112→    }
   113→  }
   114→
   115→  const fetchProducts = async (supplierId: string) => {
   116→    setLoadingProducts(true)
   117→    try {
   118→      const res = await fetch(`/api/suppliers/${supplierId}/products`)
   119→      const data = await res.json()
   120→      setProducts(data)
   121→    } catch {
   122→      setProducts([])
   123→    } finally {
   124→      setLoadingProducts(false)
   125→    }
   126→  }
   127→
   128→  useEffect(() => { fetchSuppliers(); fetchCategories() }, [])
   129→  useEffect(() => {
   130→    if (selectedId) fetchProducts(selectedId)
   131→    else setProducts([])
   132→  }, [selectedId])
   133→
   134→  const filtered = useMemo(() => {
   135→    if (!search) return suppliers
   136→    const q = search.toLowerCase()
   137→    return suppliers.filter((s) =>
   138→      s.companyName.toLowerCase().includes(q) ||
   139→      (s.city || '').toLowerCase().includes(q) ||
   140→      (s.category || '').toLowerCase().includes(q)
   141→    )
   142→  }, [suppliers, search])
   143→
   144→  const selected = suppliers.find((s) => s.id === selectedId) || null
   145→
   146→  const stats = useMemo(() => {
   147→    const total = suppliers.length
   148→    const verified = suppliers.filter((s) => s.verified).length
   149→    const productsTotal = suppliers.reduce((acc, s) => acc + (s.products?.length || 0), 0)
   150→    const avgRating = total ? suppliers.reduce((acc, s) => acc + (s.rating || 0), 0) / total : 0
   151→    return { total, verified, productsTotal, avgRating }
   152→  }, [suppliers])
   153→
   154→  const allCategories = useMemo(() => {
   155→    const flat: Category[] = []
   156→    categories.forEach((c) => {
   157→      flat.push(c)
   158→      if (c.children) c.children.forEach((ch) => flat.push(ch))
   159→    })
   160→    return flat
   161→  }, [categories])
   162→
   163→  const handleCreateSupplier = async () => {
   164→    if (!supplierForm.companyName || !supplierForm.email) {
   165→      toast('Empresa y email son obligatorios', 'error')
   166→      return
   167→    }
   168→    setSubmitting(true)
   169→    try {
   170→      const res = await fetch('/api/suppliers', {
   171→        method: 'POST',
   172→        headers: { 'Content-Type': 'application/json' },
   173→        body: JSON.stringify(supplierForm),
   174→      })
   175→      if (!res.ok) throw new Error()
   176→      await fetchSuppliers()
   177→      setNewSupplierOpen(false)
   178→      setSupplierForm({ companyName: '', email: '', phone: '', taxId: '', address: '', city: '', category: '' })
   179→      toast('Proveedor creado correctamente')
   180→    } catch {
   181→      toast('Error al crear proveedor', 'error')
   182→    } finally {
   183→      setSubmitting(false)
   184→    }
   185→  }
   186→
   187→  const handleCreateProduct = async () => {
   188→    if (!selectedId) return
   189→    if (!productForm.name || !productForm.price || !productForm.categoryId) {
   190→      toast('Nombre, precio y categoría son obligatorios', 'error')
   191→      return
   192→    }
   193→    setSubmitting(true)
   194→    try {
   195→      const payload = {
   196→        name: productForm.name,
   197→        price: parseFloat(productForm.price),
   198→        compareAtPrice: productForm.compareAtPrice ? parseFloat(productForm.compareAtPrice) : null,
   199→        stock: parseInt(productForm.stock || '0'),
   200→        sku: productForm.sku || null,
   201→        categoryId: productForm.categoryId,
   202→        supplierId: selectedId,
   203→        description: productForm.description,
   204→        imageUrl: productForm.imageUrl || null,
   205→      }
   206→      const res = await fetch('/api/products', {
   207→        method: 'POST',
   208→        headers: { 'Content-Type': 'application/json' },
   209→        body: JSON.stringify(payload),
   210→      })
   211→      if (!res.ok) throw new Error()
   212→      await fetchProducts(selectedId)
   213→      await fetchSuppliers()
   214→      setNewProductOpen(false)
   215→      setProductForm({ name: '', price: '', compareAtPrice: '', stock: '', sku: '', categoryId: '', description: '', imageUrl: '' })
   216→      toast('Producto agregado correctamente')
   217→    } catch {
   218→      toast('Error al agregar producto', 'error')
   219→    } finally {
   220→      setSubmitting(false)
   221→    }
   222→  }
   223→

   226→    const url = URL.createObjectURL(blob)

   230→    document.body.appendChild(a)
   231→    a.click()
   232→    document.body.removeChild(a)
   233→    URL.revokeObjectURL(url)
   234→  }

   242→      setCsvText(text)
   243→      toast(`Archivo "${file.name}" cargado`)

   245→    reader.readAsText(file)
   246→    e.target.value = ''
   247→  }
   248→
   249→  const handleBulk = async () => {
   250→    if (!selectedId) return
   251→    if (!csvText.trim()) {
   252→      toast('Pega o sube un CSV primero', 'error')
   253→      return

   257→      const res = await fetch('/api/products/bulk', {
   258→        method: 'POST',
   259→        headers: { 'Content-Type': 'application/json' },
   260→        body: JSON.stringify({ csv: csvText, supplierId: selectedId }),
   261→      })
   262→      const data = await res.json()
   263→      if (!res.ok) throw new Error()
   264→      await fetchProducts(selectedId)
   265→      await fetchSuppliers()
   266→      setBulkOpen(false)
   267→      setCsvText('')
   268→      toast(`Importación: ${data.created} creados, ${data.errors?.length || 0} errores de ${data.total}`, data.errors?.length ? 'info' : 'success')
   269→    } catch {
   270→      toast('Error en la carga masiva', 'error')
   271→    } finally {
   272→      setSubmitting(false)
   273→    }
   274→  }
   275→
   276→  return (
   277→    <div className="space-y-6">
   278→      {/* Header */}
   279→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   280→        <div>
   281→          <h1 className="text-2xl font-bold tracking-tight text-gradient">Proveedores</h1>
   282→          <p className="text-sm text-muted-foreground">Gestiona tu red de proveedores y su catálogo de productos.</p>
   283→        </div>
   284→        <Button onClick={() => setNewSupplierOpen(true)} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   285→          <Plus className="size-4" /> Nuevo proveedor
   286→        </Button>
   287→      </div>
   288→
   289→      {/* Stats */}
   290→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   291→        <StatCard icon={<Building2 className="size-5" />} label="Total proveedores" value={stats.total} accent="emerald" />
   292→        <StatCard icon={<ShieldCheck className="size-5" />} label="Verificados" value={stats.verified} accent="emerald" />
   293→        <StatCard icon={<Boxes className="size-5" />} label="Productos totales" value={stats.productsTotal} accent="amber" />
   294→        <StatCard icon={<Star className="size-5" />} label="Rating promedio" value={stats.avgRating.toFixed(2)} accent="amber" />
   295→      </div>
   296→
   297→      {/* Main grid */}
   298→      <div className="grid lg:grid-cols-[320px_1fr] gap-4">
   299→        {/* List */}
   300→        <Card className="lg:sticky lg:top-4 h-fit">
   301→          <CardHeader className="pb-3">
   302→            <div className="flex items-center justify-between">
   303→              <CardTitle className="text-base">Listado</CardTitle>
   304→              <Badge variant="secondary" className="bg-emerald-50 text-emerald-700">{filtered.length}</Badge>
   305→            </div>
   306→            <div className="relative mt-1">
   307→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   308→              <Input
   309→                placeholder="Buscar proveedor…"
   310→                className="pl-8 h-9"
   311→                value={search}
   312→                onChange={(e) => setSearch(e.target.value)}
   313→              />
   314→            </div>
   315→          </CardHeader>
   316→          <CardContent className="pt-0">
   317→            {loading ? (
   318→              <div className="flex items-center justify-center py-10 text-muted-foreground">
   319→                <Loader2 className="size-5 animate-spin" />
   320→              </div>
   321→            ) : filtered.length === 0 ? (
   322→              <div className="text-center py-10 text-sm text-muted-foreground">
   323→                <Building2 className="size-8 mx-auto mb-2 opacity-40" />
   324→                Sin proveedores
   325→              </div>
   326→            ) : (
   327→              <ul className="max-h-[70vh] overflow-y-auto scroll-thin -mx-2 px-2 space-y-1">
   328→                {filtered.map((s) => (
   329→                  <li key={s.id}>
   330→                    <button
   331→                      onClick={() => setSelectedId(s.id)}
   332→                      className={cn(
   333→                        'w-full text-left rounded-lg border p-3 transition-colors',
   334→                        s.id === selectedId
   335→                          ? 'border-emerald-500 bg-emerald-50/60'
   336→                          : 'border-transparent hover:bg-muted/60'
   337→                      )}
   338→                    >
   339→                      <div className="flex items-start justify-between gap-2">
   340→                        <div className="min-w-0 flex-1">
   341→                          <div className="flex items-center gap-1.5">
   342→                            <span className="font-medium text-sm truncate">{s.companyName}</span>
   343→                            {s.verified && <CheckCircle2 className="size-3.5 text-emerald-600 shrink-0" />}
   344→                          </div>
   345→                          <div className="flex items-center gap-2 text-xs text-muted-foreground mt-0.5">
   346→                            <span className="inline-flex items-center gap-0.5"><MapPin className="size-3" />{s.city || '—'}</span>
   347→                          </div>
   348→                        </div>
   349→                        {s.category && (
   350→                          <Badge variant="outline" className="text-[10px] px-1.5 py-0 border-amber-300 text-amber-700 bg-amber-50 shrink-0">
   351→                            {s.category}
   352→                          </Badge>
   353→                        )}
   354→                      </div>
   355→                      <div className="flex items-center justify-between mt-2">
   356→                        <Stars value={s.rating || 0} />
   357→                        <span className="text-xs text-muted-foreground inline-flex items-center gap-1">
   358→                          <Package className="size-3" /> {s.products?.length || 0}
   359→                        </span>
   360→                      </div>
   361→                    </button>
   362→                  </li>
   363→                ))}
   364→              </ul>
   365→            )}
   366→          </CardContent>
   367→        </Card>
   368→
   369→        {/* Detail */}
   370→        <Card>
   371→          {selected ? (
   372→            <>
   373→              <CardHeader className="border-b bg-gradient-to-br from-emerald-50/60 to-amber-50/40">
   374→                <div className="flex flex-col sm:flex-row sm:items-start sm:justify-between gap-3">
   375→                  <div>
   376→                    <div className="flex items-center gap-2 flex-wrap">
   377→                      <CardTitle className="text-xl">{selected.companyName}</CardTitle>
   378→                      {selected.verified ? (
   379→                        <Badge className="bg-emerald-600 text-white hover:bg-emerald-600">
   380→                          <CheckCircle2 className="size-3" /> Verificado
   381→                        </Badge>
   382→                      ) : (
   383→                        <Badge variant="outline" className="text-muted-foreground">No verificado</Badge>
   384→                      )}
   385→                      {selected.category && (
   386→                        <Badge variant="outline" className="border-amber-300 text-amber-700 bg-amber-50">
   387→                          {selected.category}
   388→                        </Badge>
   389→                      )}
   390→                    </div>
   391→                    <div className="mt-2"><Stars value={selected.rating || 0} /></div>
   392→                  </div>
   393→                  <div className="flex flex-wrap gap-2">
   394→                    <Button onClick={() => setNewProductOpen(true)} size="sm" className="bg-emerald-600 hover:bg-emerald-700 text-white">
   395→                      <Plus className="size-4" /> Agregar producto
   396→                    </Button>
   397→                    <Button onClick={() => setBulkOpen(true)} size="sm" variant="outline" className="border-amber-400 text-amber-700 hover:bg-amber-50">
   398→                      <Upload className="size-4" /> Carga masiva CSV
   399→                    </Button>
   400→                  </div>
   401→                </div>
   402→                <div className="grid sm:grid-cols-2 lg:grid-cols-4 gap-3 mt-4 text-sm">
   403→                  <InfoLine icon={<FileText className="size-3.5" />} label="RUC / Tax ID" value={selected.taxId || '—'} />
   404→                  <InfoLine icon={<Phone className="size-3.5" />} label="Teléfono" value={selected.phone || '—'} />
   405→                  <InfoLine icon={<MapPin className="size-3.5" />} label="Dirección" value={selected.address || '—'} />
   406→                  <InfoLine icon={<Building2 className="size-3.5" />} label="Ciudad" value={selected.city || '—'} />

   408→              </CardHeader>
   409→              <CardContent className="pt-4">
   410→                <div className="flex items-center justify-between mb-3">
   411→                  <h3 className="text-sm font-semibold flex items-center gap-2">
   412→                    <Package className="size-4 text-emerald-600" /> Productos del proveedor

   415→                </div>
   416→                {loadingProducts ? (
   417→                  <div className="flex items-center justify-center py-10 text-muted-foreground">
   418→                    <Loader2 className="size-5 animate-spin" />
   419→                  </div>
   420→                ) : products.length === 0 ? (
   421→                  <div className="text-center py-10 border border-dashed rounded-lg">
   422→                    <ImageIcon className="size-8 mx-auto mb-2 text-muted-foreground/40" />

   424→                    <Button onClick={() => setNewProductOpen(true)} size="sm" variant="outline" className="mt-3">
   425→                      <Plus className="size-4" /> Agregar el primero

   427→                  </div>

   434→                          <TableHead>Nombre</TableHead>
   435→                          <TableHead>SKU</TableHead>
   436→                          <TableHead className="text-right">Precio</TableHead>
   437→                          <TableHead className="text-center">Stock</TableHead>
   438→                          <TableHead className="text-center">Estado</TableHead>
   439→                        </TableRow>
   440→                      </TableHeader>
   441→                      <TableBody>
   442→                        {products.map((p) => (

   444→                            <TableCell>
   445→                              <img
   446→                                src={p.imageUrl || PLACEHOLDER_IMG}
   447→                                alt={p.name}
   448→                                className="size-10 rounded-md object-cover border bg-muted"
   449→                                loading="lazy"
   450→                              />

   452→                            <TableCell className="font-medium max-w-[260px] truncate">{p.name}</TableCell>
   453→                            <TableCell className="text-muted-foreground font-mono text-xs">{p.sku || '—'}</TableCell>
   454→                            <TableCell className="text-right">

   479→                    </Table>

   484→          ) : (
   485→            <CardContent className="py-20">
   486→              <div className="text-center max-w-md mx-auto">
   487→                <div className="size-16 rounded-2xl bg-emerald-50 mx-auto flex items-center justify-center mb-4">
   488→                  <Building2 className="size-8 text-emerald-600" />
   489→                </div>
   490→                <h3 className="text-lg font-semibold">Selecciona un proveedor</h3>
