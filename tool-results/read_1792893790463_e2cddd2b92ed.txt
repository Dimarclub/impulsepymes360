   356→function QRCard({ qr, websiteUrl }: { qr: NonNullable<Provider['qrCodes']>[number]; websiteUrl: string }) {
   357→  const targetUrl = qr.targetUrl?.startsWith('http')

   360→  const qrImg = `https://api.qrserver.com/v1/create-qr-code/?data=${encodeURIComponent(targetUrl)}&size=160x160`
   361→  return (
   362→    <Card className="overflow-hidden">
   363→      <CardContent className="p-4 flex gap-4">
   364→        <div className="shrink-0">
   365→          <div className="size-24 rounded-lg bg-white border p-1.5">
   366→            { }
   367→            <img src={qrImg} alt={`QR ${qr.code}`} className="size-full object-contain" />
   368→          </div>
   369→        </div>
   370→        <div className="min-w-0 flex-1">
   371→          <div className="flex items-center gap-2 flex-wrap">

   373→            <Badge className={QR_TYPE_COLORS[qr.type] || 'bg-muted text-muted-foreground'}>

   399→// ---------- Componente principal ----------
   400→
   401→export function ImpulseModule() {
   402→  const setModule = useAppStore((s) => s.setModule)
   403→  const toast = useAppStore((s) => s.toast)
   404→  const [provider, setProvider] = useState<Provider | null | undefined>(undefined) // undefined=loading, null=no hay
   405→  const [editions, setEditions] = useState<AdEdition[]>([])
   406→  const [authChecked, setAuthChecked] = useState(false)
   407→  const [refreshing, setRefreshing] = useState(false)
   408→
   409→  const fetchData = useCallback(async (silent = false) => {
   410→    if (!silent) setRefreshing(true)
   411→    try {
   412→      const [meRes, edRes] = await Promise.all([
   413→        fetch('/api/impulse/me', { cache: 'no-store' }),
   414→        fetch('/api/impulse/editions', { cache: 'no-store' }),
   415→      ])
   416→      if (meRes.status === 401) {
   417→        setAuthChecked(true)
   418→        setProvider(null)
   419→        return
   420→      }
   421→      if (meRes.ok) {
   422→        const me = await meRes.json()
   423→        setProvider(me || null)
   424→        setAuthChecked(true)
   425→      } else {
   426→        setProvider(null)
   427→        setAuthChecked(true)
   428→      }
   429→      if (edRes.ok) {
   430→        const eds = await edRes.json()
   431→        setEditions(Array.isArray(eds) ? eds : [])
   432→      }
   433→    } catch {
   434→      if (!silent) toast('No se pudo cargar el panel', 'error')
   435→    } finally {
   436→      setRefreshing(false)
   437→    }
   438→  }, [toast])
   439→
   440→  useEffect(() => {
   441→    fetchData()
   442→  }, [fetchData])
   443→
   444→  // Loading inicial
   445→  if (provider === undefined && !authChecked) {
   446→    return (
   447→      <div className="space-y-4">
   448→        <HeroHeader loading />
   449→        <LoadingSkeleton />
   450→      </div>
   451→    )
   452→  }
   453→
   454→  // No autenticado
   455→  if (authChecked && provider === null && editions.length === 0) {
   456→    return (
   457→      <div className="space-y-4">
   458→        <HeroHeader loading={false} />
   459→        <UnauthenticatedState />
   460→      </div>
   461→    )
   462→  }
   463→
   464→  // Sin provider pero autenticado (puede ser CLIENT, AFFILIATE, etc.)
   465→  if (provider === null) {
   466→    return (
   467→      <div className="space-y-4">
   468→        <HeroHeader loading={false} />
   469→        <NoProviderState />
   470→      </div>
   471→    )
   472→  }
   473→
   474→  // Provider activo
   475→  const plan = provider.subscription?.plan
   476→  const isActive = (e: ElementoImpulse) =>
   477→    e.alwaysActive || (e.flag ? Boolean((plan as any)?.[e.flag]) : true)
   478→
   479→  const totalQrScans = (provider.qrCodes || []).reduce((sum, q) => sum + (q.scans || 0), 0)
   480→  const webViews = provider.website?.views || 0
   481→  const adSpacesCount = (provider.adSpaces || []).length
   482→
   483→  return (
   484→    <div className="space-y-6">
   485→      {/* Hero header */}
   486→      <HeroHeader
   487→        loading={false}
   488→        onRefresh={() => fetchData(true)}
   489→        refreshing={refreshing}
   490→      />
   491→
   492→      {/* 5 mini-cards de elementos */}
   493→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-3">
   494→        {ELEMENTOS.map((el, i) => (
   495→          <MiniElementCard
   496→            key={el.key}
   497→            icon={el.icon}
   498→            name={el.name}
   499→            status={isActive(el) ? 'active' : 'inactive'}
   500→            index={i}
   501→          />
   502→        ))}
   503→      </div>
   504→
   505→      {/* Stats row */}
   506→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   507→        <StatCard icon={Package} label="Productos publicados" value="—" accent="emerald" />
   508→        <StatCard icon={Eye} label="Vistas Web Express" value={webViews} accent="amber" />
   509→        <StatCard icon={QrCode} label="Escaneos QR totales" value={totalQrScans} accent="rose" />
   510→        <StatCard icon={ShoppingBag} label="Espacios PubliBags" value={adSpacesCount} accent="muted" />
   511→      </div>
   512→
   513→      {/* Grid principal: provider + subscription */}
   514→      <div className="grid lg:grid-cols-2 gap-4">
   515→        {/* Provider info card */}
   516→        <Card className="overflow-hidden">
   517→          <CardHeader className="bg-gradient-to-br from-primary/5 to-accent/5 pb-4">
   518→            <div className="flex items-start justify-between gap-2">
   519→              <CardTitle className="text-base flex items-center gap-2">
   520→                <Store className="size-4 text-emerald-600" />
   521→                Información del negocio
   522→              </CardTitle>
   523→              <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={() => setModule('impulse-wizard')}>
   524→                <Settings2 className="size-3.5 mr-1" />
   525→                Editar
   526→              </Button>
   527→            </div>
   528→          </CardHeader>
   529→          <CardContent className="p-4 space-y-4">
   530→            <div className="flex items-start gap-4">
   531→              <div className="size-16 rounded-xl bg-muted overflow-hidden shrink-0 border">
   532→                {provider.logoUrl ? (
   533→                   
   534→                  <img src={provider.logoUrl} alt={provider.companyName} className="size-full object-cover" />
   535→                ) : (
   536→                  <div className="size-full grid place-items-center text-muted-foreground">

   539→                )}
   540→              </div>
   541→              <div className="min-w-0 flex-1">
   542→                <div className="flex items-center gap-2 flex-wrap">
   543→                  <h3 className="text-lg font-bold text-foreground truncate">{provider.companyName}</h3>

   547→                </div>

   549→                  {provider.category || 'Sin categoría'}
   550→                </div>
