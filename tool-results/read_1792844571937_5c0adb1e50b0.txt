   493→// ===================== PLANO LIVE =====================
   494→function PlanoLive({ data, loading, onBuy }: { data: BagPlanData | null; loading: boolean; onBuy: () => void }) {
   495→  const [view, setView] = useState<'current' | 'next'>('current')
   496→
   497→  const edition = view === 'current' ? data?.currentEdition : data?.nextEdition
   498→  const spaces = view === 'current' ? data?.currentSpaces : data?.nextSpaces
   499→  const stats = view === 'current' ? data?.currentStats : data?.nextStats
   500→
   501→  // Para la campaña actual (todos vendidos), agrupamos sponsors para la leyenda
   502→  const sponsors = useMemo(() => {
   503→    if (!spaces) return []
   504→    const map = new Map<string, number>()
   505→    for (const s of spaces) {
   506→      if (s.status !== 'SOLD') continue
   507→      const n = s.buyerName || ''
   508→      if (!n) continue
   509→      map.set(n, (map.get(n) || 0) + 1)
   510→    }
   511→    return Array.from(map.entries()).sort((a, b) => b[1] - a[1]).slice(0, 12)
   512→  }, [spaces])
   513→
   514→  return (
   515→    <Section

   655→function LivePanel({ panel, spaces, view }: { panel: BagPanel; spaces: BagSpace[]; view: 'current' | 'next' }) {
   656→  const byPos = useMemo(() => {
   657→    const m = new Map<number, BagSpace>()
   658→    for (const s of spaces) m.set(s.position, s)
   659→    return m
   660→  }, [spaces])
   661→
   662→  const positions: number[] = []
   663→  for (let i = panel.start; i <= panel.end; i++) positions.push(i)
   664→
   665→  return (
   666→    <div className="rounded-xl p-4" style={{ background: C.card, border: `1px solid ${C.border}` }}>
   667→      <div className="flex items-center justify-between mb-3">
   668→        <div>
   669→          <p style={{ fontSize: 14, fontWeight: 600, color: C.text }}>{panel.label}</p>
   670→          <p style={{ fontSize: 11, color: C.muted }}>{panel.widthCm}×{panel.heightCm} cm · {panel.cols}×{panel.rows}</p>
   671→        </div>
   672→        <span style={{ fontSize: 11, color: C.muted }}>#{panel.start}–{panel.end}</span>
   673→      </div>
   674→      <div
   675→        className="grid"
   676→        style={{ gridTemplateColumns: `repeat(${panel.cols}, minmax(0, 1fr))`, gap: 2 }}
   677→      >
   678→        {positions.map((pos) => {
   679→          const s = byPos.get(pos)
   680→          let bg = C.bg
   681→          let title = `#${pos} · Disponible`
   682→          if (view === 'current') {
   683→            // Todos vendidos → color por sponsor
   684→            const name = s?.buyerName
   685→            if (name) {
   686→              bg = sponsorColor(name)
   687→              title = `#${pos} · ${name}`
   688→            }
   689→          } else {
   690→            // next → color por status
   691→            if (s?.status === 'SOLD') {
   692→              const name = s.buyerName
   693→              bg = name ? sponsorColor(name) : C.rose
   694→              title = `#${pos} · Vendido${name ? ` · ${name}` : ''}`
   695→            } else if (s?.status === 'RESERVED') {
   696→              bg = C.amber
   697→              title = `#${pos} · Reservado`
   698→            } else {
   699→              bg = `${C.emerald}33`
   700→              title = `#${pos} · Disponible`
   701→            }
   702→          }
   703→          return (
   704→            <div
   705→              key={pos}
   706→              title={title}
   707→              style={{
   708→                aspectRatio: '1 / 1',
   709→                background: bg,
   710→                borderRadius: 2,
   711→                minWidth: 6,
   712→                cursor: 'default',
   713→              }}
   714→            />
   715→          )
   716→        })}
   717→      </div>
   718→    </div>
   719→  )
   720→}
