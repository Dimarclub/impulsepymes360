<script>
    // ==========================================
    // CONSTANTES
    // ==========================================
    var FLETE_COMISION = 0.05;

    // ==========================================
    // VARIABLES GLOBALES
    // ==========================================
    var deltas = [], pedidos = [], mensajeria = [], incidencias = [], notificaciones = [];
    var deltaActual = null;
    var firmaPendiente = null;
    var confirmCallback = null;

    // ==========================================
    // DATA MANAGEMENT
    // ==========================================
    function gD() {
        var s = getMainData();
        deltas = s.data.deltas || [];
        pedidos = s.data.pedidos || [];
        mensajeria = s.data.mensajeria || [];
        incidencias = s.data.incidencias || [];
        notificaciones = s.data.notificaciones || [];
        cargarSelect();
        actualizarBadgeNotif();
    }

    function sD() {
        try {
            var mainData = getMainData();
            mainData.data.deltas = deltas;
            mainData.data.pedidos = pedidos;
            mainData.data.mensajeria = mensajeria;
            mainData.data.incidencias = incidencias;
            mainData.data.notificaciones = notificaciones;
            saveMainData(mainData);
        } catch(e) {
            toast('Error: Memoria llena.', 'error');
        }
    }

    function getMainData() {
        try { return JSON.parse(localStorage.getItem('dc_data') || '{"data":{},"cnt":{}}'); }
        catch(e) { return { data: {}, cnt: {} }; }
    }

    function saveMainData(data) {
        try { localStorage.setItem('dc_data', JSON.stringify(data)); }
        catch(e) { toast('Error al guardar', 'error'); }
    }

    // ==========================================
    // SELECT DE DELTAS
    // ==========================================
    function cargarSelect() {
        var sel = document.getElementById('selDelta');
        var disponibles = deltas.filter(function(d) {
            return d.estado === 'Disponible' || d.estado === 'En Ruta' || d.estado === 'En Alerta';

    function rDash() {
        if (!deltaActual) return;
        gD();
        for (var i = 0; i < deltas.length; i++) {
            if (deltas[i].id === deltaActual.id) { deltaActual = deltas[i]; break; }
        }
        if (!deltaActual) return;

        var misPedidos = [];
        var misSobres = [];
        for (var j = 0; j < pedidos.length; j++) {
            if (pedidos[j].delta === deltaActual.id) misPedidos.push(pedidos[j]);
        }
        for (var k = 0; k < mensajeria.length; k++) {
            if (mensajeria[k].deltaId === deltaActual.id) misSobres.push(mensajeria[k]);
        }

        var pendPedidos = misPedidos.filter(function(p) { return p.estado !== 'Entregado / Cobrado'; });
        var pendSobres = misSobres.filter(function(m) { return m.estado !== 'Entregado / Cobrado'; });
        var totalPend = pendPedidos.length + pendSobres.length;

        var hechosPedidos = misPedidos.filter(function(p) { return p.estado === 'Entregado / Cobrado'; });
        var hechosSobres = misSobres.filter(function(m) { return m.estado === 'Entregado / Cobrado'; });
        var totalHechos = hechosPedidos.length + hechosSobres.length;

        var hoy = new Date().toISOString().split('T')[0];
        var entregasHoy = 0;
        if (deltaActual.historial) {
            for (var l = 0; l < deltaActual.historial.length; l++) {
                if (deltaActual.historial[l].fecha === hoy) entregasHoy++;
            }
        }

        document.getElementById('kPend').textContent = totalPend;
        document.getElementById('kHechos').textContent = totalHechos;
        document.getElementById('kHoy').textContent = entregasHoy;

        var gananciaPedidos = 0;
        for (var m = 0; m < hechosPedidos.length; m++) {
            gananciaPedidos += hechosPedidos[m].total * FLETE_COMISION;
        }
        var gananciaSobres = 0;
        for (var n = 0; n < hechosSobres.length; n++) {
            gananciaSobres += hechosSobres[n].monto * FLETE_COMISION;
        }
        document.getElementById('kGanado').textContent = fM(gananciaPedidos + gananciaSobres);
    }

    function prepararEntrega(cod, tipo) {
        firmaPendiente = { cod: cod, tipo: tipo };

        document.getElementById('firmaInput').value = '';

    function confirmarFirma() {
        var firma = document.getElementById('firmaInput').value.trim();
        if (!firma) { toast('Ingresa la firma del receptor', 'error'); return; }
        if (!firmaPendiente) return;
        var cod = firmaPendiente.cod;
        var tipo = firmaPendiente.tipo;

        if (tipo === 'pedido') {
            var p = null;
            for (var i = 0; i < pedidos.length; i++) {
                if (pedidos[i].cod === cod) { p = pedidos[i]; break; }
            }
            if (!p) { toast('Pedido no encontrado', 'error'); cerrarModal('moFirma'); return; }
            if (p.estado === 'Entregado / Cobrado') { toast('Este pedido ya fue entregado', 'error'); cerrarModal('moFirma'); return; }
            p.estado = 'Entregado / Cobrado';
            if (!p.fecha) p.fecha = new Date().toISOString().split('T')[0];

            var mainData = getMainData();
            if (!mainData.data) mainData.data = {};
            if (!mainData.cnt) mainData.cnt = {};
            if (!mainData.data.ingresos) mainData.data.ingresos = [];
            if (!mainData.cnt.i) mainData.cnt.i = 1;
            mainData.data.ingresos.push({
                id: 'ING-' + String(mainData.cnt.i++).padStart(4, '0'),
                fec: new Date().toISOString().split('T')[0],
                con: 'Cobro Delta App - Pedido ' + p.cod + ' (' + p.cliente + ')',
                mon: p.total,
                cat: 'Ventas Marketplace'
            });
            saveMainData(mainData);

            var d = null;
            for (var j = 0; j < deltas.length; j++) {
                if (deltas[j].id === deltaActual.id) { d = deltas[j]; break; }
            }
            if (d) {
                if (!d.historial) d.historial = [];
                d.historial.push({ fecha: new Date().toISOString().split('T')[0], cod: cod, tipo: 'pedido', firma: firma });
                d.estado = 'Disponible';
                d.entregas = (d.entregas || 0) + 1;
            }
            toast('✅ Pedido entregado. Ganancia: ' + fM(p.total * FLETE_COMISION), 'success');
            agregarNotificacion('Pedido ' + cod + ' entregado a ' + p.cliente, 'success');

        } else if (tipo === 'sobre') {
            var m = null;
            for (var k = 0; k < mensajeria.length; k++) {
                if (mensajeria[k].id === cod) { m = mensajeria[k]; break; }
            }
            if (!m) { toast('Sobre no encontrado', 'error'); cerrarModal('moFirma'); return; }
            if (m.estado === 'Entregado / Cobrado') { toast('Este sobre ya fue entregado', 'error'); cerrarModal('moFirma'); return; }
            m.estado = 'Entregado / Cobrado';

            if (m.monto > 0) {
                var mainData2 = getMainData();
                if (!mainData2.data) mainData2.data = {};
                if (!mainData2.cnt) mainData2.cnt = {};
                if (!mainData2.data.ingresos) mainData2.data.ingresos = [];
                if (!mainData2.cnt.i) mainData2.cnt.i = 1;
                mainData2.data.ingresos.push({
                    id: 'ING-' + String(mainData2.cnt.i++).padStart(4, '0'),
                    fec: new Date().toISOString().split('T')[0],
                    con: 'Mensajería B2B - Sobre ' + m.id + ' (' + m.empresa + ')',
                    mon: m.monto,
                    cat: 'Mensajería B2B'
                });
                saveMainData(mainData2);
            }

            var d2 = null;
            for (var l = 0; l < deltas.length; l++) {
                if (deltas[l].id === deltaActual.id) { d2 = deltas[l]; break; }
            }
            if (d2) {
                if (!d2.historial) d2.historial = [];
                d2.historial.push({ fecha: new Date().toISOString().split('T')[0], cod: cod, tipo: 'sobre', firma: firma });
                d2.estado = 'Disponible';
                d2.entregas = (d2.entregas || 0) + 1;
            }
            toast('✅ Sobre entregado. Ganancia: ' + fM(m.monto * FLETE_COMISION), 'success');
            agregarNotificacion('Sobre ' + cod + ' entregado a ' + m.destinatario, 'success');
        }

        sD();
        cerrarModal('moFirma');
        firmaPendiente = null;
        rDash();
        rViajes();
        cargarSelect();
    }

    function guardarIncidenciaDelta() {
        var raw = document.getElementById('incidenciaCod').value;
        var tipoInc = document.getElementById('incTipo').value;
        var desc = document.getElementById('incDesc').value.trim();
        if (!desc) { toast('Describe la incidencia', 'error'); return; }
        var parts = raw.split('|');
        var cod = parts[0];
        var tipo = parts[1] || 'general';

        incidencias.push({
            id: 'INC-' + String(incidencias.length + 1).padStart(4, '0'),
            codigo: cod,
            tipo: tipo,
            incTipo: tipoInc,
            desc: desc,
            fecha: new Date().toISOString().split('T')[0],
            delta: deltaActual.id,
            resuelta: false
        });

        if (tipo === 'pedido') {
            for (var i = 0; i < pedidos.length; i++) {
                if (pedidos[i].cod === cod) { pedidos[i].estado = 'Pendiente (Alerta Master)'; break; }
            }
        }
        sD();
        cerrarModal('moIncidencia');
        toast('Incidencia reportada', 'success');
        agregarNotificacion('Incidencia reportada para ' + cod, 'info');
        rViajes();
    }

    function agregarNotificacion(msg, tipo) {
        notificaciones.push({
            cod: deltaActual ? deltaActual.id : 'sistema',

            fecha: new Date().toISOString().split('T')[0],

    function actualizarBadgeNotif() {
        var badge = document.getElementById('notifBadge');
        var noLeidas = 0;
        for (var i = 0; i < notificaciones.length; i++) {
            if (!notificaciones[i].leida && notificaciones[i].cod === (deltaActual ? deltaActual.id : 'sistema')) {
                noLeidas++;
            }
        }
        badge.textContent = noLeidas;
        badge.style.display = noLeidas > 0 ? 'inline' : 'none';
    }

    function abrirNotificaciones() {
        var list = document.getElementById('notifList');
        var misNotifs = [];
        for (var i = 0; i < notificaciones.length; i++) {
            if (notificaciones[i].cod === (deltaActual ? deltaActual.id : 'sistema')) {
                misNotifs.push(notificaciones[i]);
            }
        }
        if (!misNotifs.length) {
            list.innerHTML = '<div class="empty-state"><i class="fas fa-bell-slash"></i><p>Sin notificaciones</p></div>';
        } else {
            var html = '';
            for (var j = misNotifs.length - 1; j >= 0; j--) {
                var n = misNotifs[j];
                var tipoIcon = n.tipo === 'success' ? 'fa-circle-check' : n.tipo === 'error' ? 'fa-circle-exclamation' : 'fa-circle-info';
                html += '<div class="notif-item">' +
                    '<div class="notif-icon ' + n.tipo + '"><i class="fas ' + tipoIcon + '"></i></div>' +
                    '<div class="notif-text"><p>' + n.msg + '</p><span>' + n.fecha + '</span></div>' +
                '</div>';
            }
            list.innerHTML = html;
            for (var k = 0; k < notificaciones.length; k++) {
                if (notificaciones[k].cod === deltaActual.id) notificaciones[k].leida = true;
            }
            sD();
            actualizarBadgeNotif();
        }
        abrirModal('moNotif');
    }
