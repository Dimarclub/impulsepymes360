function gD() {
    const s = JSON.parse(localStorage.getItem('dc_data') || '{"data":{},"cnt":{}}');

    deltas = s.data.deltas || [];

    incidencias = s.data.incidencias || [];

}

function sD() {
    try {
        const mainData = JSON.parse(localStorage.getItem('dc_data') || '{"data":{},"cnt":{}}');
        if(!mainData.data) mainData.data = {};
        if(!mainData.cnt) mainData.cnt = {};
        mainData.data.deltas = deltas;
        mainData.data.pedidos = pedidos;
        mainData.data.mensajeria = mensajeria;
        mainData.data.incidencias = incidencias;
        mainData.data.notificaciones = notificaciones;
        localStorage.setItem('dc_data', JSON.stringify(mainData));
    } catch(e) {
        toast('Error: Memoria llena.', 'error');
    }
}

function cargarSelect() {
    const sel = document.getElementById('selDelta');
    const disponibles = deltas.filter(d => d.estado === 'Disponible' || d.estado === 'En Ruta' || d.estado === 'En Alerta');
    sel.innerHTML = '<option value="">-- Selecciona tu ID --</option>';
    disponibles.forEach(function(d) {
        const o = document.createElement('option');
        o.value = d.id;
        o.textContent = d.id + ' - ' + d.nom + ' (' + d.estado + ')';
        if(deltaActual && d.id === deltaActual.id) o.selected = true;
        sel.appendChild(o);
    });
}

            <h3>${deltaActual.nom} <span class="estado-indicator ${estadoClase}" style="margin-left:8px"><i class="fas fa-circle" style="font-size:.5rem"></i> ${deltaActual.estado}</span></h3>
            <p><i class="fas fa-id-badge" style="color:var(--accent);margin-right:3px"></i> ${deltaActual.id} &nbsp;|&nbsp; 
               <i class="fas fa-motorcycle" style="color:var(--blue);margin-right:3px"></i> ${deltaActual.veh} ${deltaActual.placa !== 'N/A' ? '('+deltaActual.placa+')' : ''} &nbsp;|&nbsp;
               <i class="fas fa-clock" style="color:var(--gold);margin-right:3px"></i> ${deltaActual.turno || 'N/A'}
            </p>
        </div>`;
    rDash();
    rViajes();
}

function rDash() {
    if(!deltaActual) return;
    gD(); 
    deltaActual = deltas.find(d => d.id === deltaActual.id); 
    if(!deltaActual) return;
    const misPedidos = pedidos.filter(p => p.delta === deltaActual.id);
    const misSobres = mensajeria.filter(m => m.deltaId === deltaActual.id);
    const pendPedidos = misPedidos.filter(p => p.estado !== 'Entregado / Cobrado');
    const pendSobres = misSobres.filter(m => m.estado !== 'Entregado / Cobrado');
    const totalPend = pendPedidos.length + pendSobres.length;
    const hechosPedidos = misPedidos.filter(p => p.estado === 'Entregado / Cobrado');
    const hechosSobres = misSobres.filter(m => m.estado === 'Entregado / Cobrado');
    const totalHechos = hechosPedidos.length + hechosSobres.length;
    const hoy = gFecha();
    const entregasHoy = (deltaActual.historial || []).filter(h => h.fecha === hoy).length;
    document.getElementById('kPend').textContent = totalPend;
    document.getElementById('kHechos').textContent = totalHechos;
    document.getElementById('kHoy').textContent = entregasHoy;
    const gananciaPedidos = hechosPedidos.reduce((s, p) => s + (p.total * FLETE_COMISION), 0);
    const gananciaSobres = hechosSobres.reduce((s, m) => s + (m.monto * FLETE_COMISION), 0);
    document.getElementById('kGanado').textContent = fM(gananciaPedidos + gananciaSobres);
}

function rViajes() {
    if(!deltaActual) return;
    const b = document.getElementById('viajesB');
    const fTipo = document.getElementById('fTipoTarea').value;
    let pendPedidos = pedidos.filter(p => p.delta === deltaActual.id && p.estado !== 'Entregado / Cobrado');
    let pendSobres = mensajeria.filter(m => m.deltaId === deltaActual.id && m.estado !== 'Entregado / Cobrado');
    if(fTipo === 'pedido') pendSobres = [];
    if(fTipo === 'sobre') pendPedidos = [];
    const totalPend = pendPedidos.length + pendSobres.length;
    if(!totalPend) {
        b.innerHTML = '<tr><td colspan="6"><div class="empty-s" style="padding:24px"><i class="fas fa-check-double"></i><p>¡Excelente! No tienes tareas pendientes.</p></div></td></tr>';
    } else {
        let html = '';
        pendPedidos.forEach(p => {
            html += `<tr><td style="color:var(--accent);font-weight:700;white-space:nowrap">${p.cod}</td><td><span class="rb rb-pend">Pedido</span></td><td><strong>${escHtml(p.cliente)}</strong></td><td style="font-size:.72rem;color:var(--muted)">${escHtml(p.dir)}</td><td style="text-align:right;font-weight:700;color:var(--gold);white-space:nowrap">${fM(p.total)}</td><td style="text-align:center"><button class="btn btn-p btn-sm" onclick="prepararEntrega('${p.cod}','pedido')"><i class="fas fa-check"></i> Entregar</button> <button class="btn btn-sm btn-d" onclick="reportarIncidencia('${p.cod}','pedido')"><i class="fas fa-flag"></i></button></td></tr>`;
        });
        pendSobres.forEach(m => {
            html += `<tr><td style="color:var(--purple);font-weight:700;white-space:nowrap">${m.id}</td><td><span class="rb rb-ruta">Sobre B2B</span></td><td><strong>${escHtml(m.empresa)}</strong><br><span style="font-size:.68rem;color:var(--muted)">${escHtml(m.destinatario)}</span></td><td style="font-size:.72rem;color:var(--muted)">${escHtml(m.dir)}</td><td style="text-align:right;font-weight:700;color:var(--gold);white-space:nowrap">${fM(m.monto)}</td><td style="text-align:center"><button class="btn btn-p btn-sm" onclick="prepararEntrega('${m.id}','sobre')"><i class="fas fa-check"></i> Entregar</button> <button class="btn btn-sm btn-d" onclick="reportarIncidencia('${m.id}','sobre')"><i class="fas fa-flag"></i></button></td></tr>`;
        });
        b.innerHTML = html;
    }
    const hB = document.getElementById('histB');
    const hechosPedidos = pedidos.filter(p => p.delta === deltaActual.id && p.estado === 'Entregado / Cobrado');
    const hechosSobres = mensajeria.filter(m => m.deltaId === deltaActual.id && m.estado === 'Entregado / Cobrado');
    if(!hechosPedidos.length && !hechosSobres.length) {
        hB.innerHTML = '<tr><td colspan="5"><div class="empty-s" style="padding:24px"><p>Sin historial de entregas aún.</p></div></td></tr>';
    } else {
        let histHtml = '';
        hechosPedidos.slice().reverse().forEach(p => {
            const ganancia = p.total * FLETE_COMISION;
            histHtml += `<tr><td style="color:var(--accent);font-weight:700">${p.cod}</td><td><span class="rb rb-ok">Pedido</span></td><td>${escHtml(p.cliente)}</td><td style="font-size:.72rem;color:var(--muted)">${p.fecha || gFecha()}</td><td style="text-align:right;font-weight:700;color:var(--success)">${fM(ganancia)}</td></tr>`;
        });
        hechosSobres.slice().reverse().forEach(m => {
            const ganancia = m.monto * FLETE_COMISION;
            histHtml += `<tr><td style="color:var(--purple);font-weight:700">${m.id}</td><td><span class="rb rb-ok">Sobre</span></td><td>${escHtml(m.empresa)}</td><td style="font-size:.72rem;color:var(--muted)">${m.fec || gFecha()}</td><td style="text-align:right;font-weight:700;color:var(--success)">${fM(ganancia)}</td></tr>`;
        });
        hB.innerHTML = histHtml;
    }
}

function prepararEntrega(cod, tipo) {
    firmaPendiente = { cod: cod, tipo: tipo };
    document.getElementById('firmaCod').textContent = cod;
    document.getElementById('firmaInput').value = '';
    abrirModal('moFirma');
}

function confirmarFirma() {
    const firma = document.getElementById('firmaInput').value.trim();
    if(!firma){ toast('Ingresa la firma del receptor', 'error'); return; }
    if(!firmaPendiente) return;
    const cod = firmaPendiente.cod;
    const tipo = firmaPendiente.tipo;
    if(tipo === 'pedido'){
        const p = pedidos.find(x => x.cod === cod);
        if(!p){ toast('Pedido no encontrado', 'error'); cerrarModal('moFirma'); return; }
        if(p.estado === 'Entregado / Cobrado'){ toast('Este pedido ya fue entregado', 'error'); cerrarModal('moFirma'); return; }
        p.estado = 'Entregado / Cobrado';
        if(!p.fecha) p.fecha = gFecha();
        const mainData = JSON.parse(localStorage.getItem('dc_data') || '{"data":{},"cnt":{}}');
        if(!mainData.data) mainData.data = {}; if(!mainData.cnt) mainData.cnt = {};
        if(!mainData.data.ingresos) mainData.data.ingresos = [];
        const currentCntI = mainData.cnt.i || 1;
        mainData.data.ingresos.push({ id:'ING-'+String(currentCntI++).padStart(4,'0'), fec: gFecha(), con:'Cobro Delta App - Pedido '+p.cod+' ('+p.cliente+')', mon: p.total, cat: 'Ventas Marketplace' });
        mainData.cnt.i = currentCntI;
        const d = deltas.find(x => x.id === deltaActual.id);
        if(d) { if(!d.historial) d.historial = []; d.historial.push({ fecha: gFecha(), cod: cod, tipo: 'pedido', firma: firma }); d.estado = 'Disponible'; d.entregas = (d.entregas || 0) + 1; }
        try { localStorage.setItem('dc_data', JSON.stringify(mainData)); } catch(e) { toast('Error al guardar ingreso', 'error'); }
        toast('✅ Pedido entregado. Ganancia: '+fM(p.total * FLETE_COMISION), 'success');
        agregarNotificacion('Pedido '+cod+' entregado a '+p.cliente, 'success');
    } else if(tipo === 'sobre'){
        const m = mensajeria.find(x => x.id === cod);
        if(!m){ toast('Sobre no encontrado', 'error'); cerrarModal('moFirma'); return; }
        if(m.estado === 'Entregado / Cobrado'){ toast('Este sobre ya fue entregado', 'error'); cerrarModal('moFirma'); return; }
        m.estado = 'Entregado / Cobrado';
        if(m.monto > 0){
            const mainData = JSON.parse(localStorage.getItem('dc_data') || '{"data":{},"cnt":{}}');
            if(!mainData.data) mainData.data = {}; if(!mainData.cnt) mainData.cnt = {};
            if(!mainData.data.ingresos) mainData.data.ingresos = [];
            const currentCntI = mainData.cnt.i || 1;
            mainData.data.ingresos.push({ id:'ING-'+String(currentCntI++).padStart(4,'0'), fec: gFecha(), con:'Mensajería B2B - Sobre '+m.id+' ('+m.empresa+')', mon: m.monto, cat: 'Mensajería B2B' });
            mainData.cnt.i = currentCntI;
            try { localStorage.setItem('dc_data', JSON.stringify(mainData)); } catch(e) { toast('Error al guardar ingreso', 'error'); }
        }
        const d = deltas.find(x => x.id === deltaActual.id);
        if(d) { if(!d.historial) d.historial = []; d.historial.push({ fecha: gFecha(), cod: cod, tipo: 'sobre', firma: firma }); d.estado = 'Disponible'; d.entregas = (d.entregas || 0) + 1; }
        toast('✅ Sobre entregado. Ganancia: '+fM(m.monto * FLETE_COMISION), 'success');
        agregarNotificacion('Sobre '+cod+' entregado a '+m.destinatario, 'success');
    }
    sD();
    cerrarModal('moFirma');
    firmaPendiente = null;
    rDash();
    rViajes();
    cargarSelect();
}

function reportarIncidencia(cod, tipo) {
    document.getElementById('incidenciaCod').value = cod + '|' + tipo;
    document.getElementById('incDesc').value = '';
    abrirModal('moIncidencia');
}

function guardarIncidenciaDelta() {
    const raw = document.getElementById('incidenciaCod').value;
    const tipoInc = document.getElementById('incTipo').value;
    const desc = document.getElementById('incDesc').value.trim();
    if(!desc){ toast('Describe la incidencia', 'error'); return; }
    const cod = raw.split('|')[0];
    const tipo = raw.split('|')[1] || 'general';
    incidencias.push({ id: 'INC-' + String(incidencias.length + 1).padStart(4,'0'), codigo: cod, tipo: tipo, incTipo: tipoInc, desc: desc, fecha: gFecha(), delta: deltaActual.id, resuelta: false });
    if(tipo === 'pedido'){
        const p = pedidos.find(x => x.cod === cod);
        if(p) p.estado = 'Pendiente (Alerta Master)';
    }
    sD();
    cerrarModal('moIncidencia');
    toast('Incidencia reportada', 'success');
    agregarNotificacion('Incidencia reportada para '+cod, 'info');
    rViajes();
}

function agregarNotificacion(msg, tipo) {
    notificaciones.push({ cod: deltaActual ? deltaActual.id : 'sistema', msg: msg, tipo: tipo, fecha: gFecha(), leida: false });
    sD();
    actualizarBadgeNotif();
}

function actualizarBadgeNotif() {
    const badge = document.getElementById('notifBadge');
    const noLeidas = notificaciones.filter(n => !n.leida && n.cod === (deltaActual ? deltaActual.id : 'sistema')).length;
    badge.textContent = noLeidas;
    badge.style.display = noLeidas > 0 ? 'inline' : 'none';
}

function abrirNotificaciones() {
    const list = document.getElementById('notifList');
    const misNotifs = notificaciones.filter(n => n.cod === (deltaActual ? deltaActual.id : 'sistema'));
    if(!misNotifs.length){
        list.innerHTML = '<div class="empty-s" style="padding:20px"><i class="fas fa-bell-slash"></i><p>Sin notificaciones</p></div>';
    } else {
        list.innerHTML = misNotifs.slice().reverse().map(n => `
            <div class="notif-item">
                <div class="notif-icon ${n.tipo}"><i class="fas ${n.tipo === 'success' ? 'fa-circle-check' : n.tipo === 'error' ? 'fa-circle-exclamation' : 'fa-circle-info'}"></i></div>
                <div class="notif-text"><p>${n.msg}</p><span>${n.fecha}</span></div>
            </div>
        `).join('');
        notificaciones.forEach(n => { if(n.cod === deltaActual.id) n.leida = true; });
        sD();
        actualizarBadgeNotif();
    }
    abrirModal('moNotif');
}
