        /**
         * Initialize State Store for 270 Slots
         */
        function initStore() {
            PANELS.forEach(panel => {
                for (let r = 0; r < panel.rows; r++) {
                    for (let c = 0; c < panel.cols; c++) {
                        const id = `${panel.id}-${r + 1}-${c + 1}`;
                        slotsData[id] = {
                            id: id,
                            zoneId: panel.id,
                            zoneName: panel.name,
                            row: r + 1,
                            col: c + 1,
                            status: 'AVAILABLE', // AVAILABLE, RESERVED, SOLD
                            company: '',
                            url: '',
                            price: standardPrice,
                            contact: ''
                        };
                    }
                }
            });
        }

        /**
         * Render Main SVG Pattern Canvas
         */
        function renderSVGBlueprint() {
            const svg = document.getElementById('svg-blueprint');
            svg.innerHTML = '';

            const rootGroup = document.createElementNS('http://www.w3.org/2000/svg', 'g');
            rootGroup.setAttribute('id', 'pattern-root');
            if (activeViewMode === 'TECHNICAL') {
                rootGroup.classList.add('mode-technical');
            }

            const startX = 40;
            const startY = 40;

            PANELS.forEach(panel => {
                const px = startX + panel.relX;
                const py = startY + panel.relY;

                const panelGroup = document.createElementNS('http://www.w3.org/2000/svg', 'g');
                panelGroup.setAttribute('class', 'panel-group');
                panelGroup.setAttribute('data-zone', panel.id);

                // Fabric Outline
                const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                rect.setAttribute('x', px);
                rect.setAttribute('y', py);
                rect.setAttribute('width', panel.width);
                rect.setAttribute('height', panel.height);
                rect.setAttribute('class', 'panel-bg');
                panelGroup.appendChild(rect);

                // Seam Outer Line (1cm seam allowance)
                const seamRect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                seamRect.setAttribute('x', px - SEAM);
                seamRect.setAttribute('y', py - SEAM);
                seamRect.setAttribute('width', panel.width + (SEAM * 2));
                seamRect.setAttribute('height', panel.height + (SEAM * 2));
                seamRect.setAttribute('class', 'seam-line');
                seamRect.setAttribute('fill', 'none');
                panelGroup.appendChild(seamRect);

                // Inner Protection Margin Line (1cm inside boundary)
                const marginRect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                marginRect.setAttribute('x', px + MARGIN);
                marginRect.setAttribute('y', py + MARGIN);
                marginRect.setAttribute('width', panel.width - (MARGIN * 2));
                marginRect.setAttribute('height', panel.height - (MARGIN * 2));
                marginRect.setAttribute('class', 'margin-line');
                marginRect.setAttribute('fill', 'none');
                panelGroup.appendChild(marginRect);

                // Panel Title Label
                const label = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                label.setAttribute('x', px + (panel.width / 2));
                label.setAttribute('y', py - 12);
                label.setAttribute('fill', panel.id === 'BASE' ? '#38bdf8' : '#94a3b8');
                label.setAttribute('font-size', '13');
                label.setAttribute('font-weight', 'bold');
                label.setAttribute('text-anchor', 'middle');
                label.textContent = `${panel.name.toUpperCase()} (${panel.width/SCALE}x${panel.height/SCALE} cm)`;
                panelGroup.appendChild(label);

                // Render Slot Grids
                const gridStartX = px + MARGIN;
                const gridStartY = py + MARGIN;

                for (let r = 0; r < panel.rows; r++) {
                    for (let c = 0; c < panel.cols; c++) {
                        const slotX = gridStartX + (c * SLOT_SIZE);
                        const slotY = gridStartY + (r * SLOT_SIZE);
                        const slotId = `${panel.id}-${r + 1}-${c + 1}`;
                        const data = slotsData[slotId];

                        const slotGroup = document.createElementNS('http://www.w3.org/2000/svg', 'g');
                        slotGroup.setAttribute('class', 'slot-item');
                        slotGroup.setAttribute('data-id', slotId);

                        const slotRect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                        slotRect.setAttribute('x', slotX);
                        slotRect.setAttribute('y', slotY);
                        slotRect.setAttribute('width', SLOT_SIZE);
                        slotRect.setAttribute('height', SLOT_SIZE);
                        slotRect.setAttribute('rx', '3');
                        slotRect.setAttribute('id', `rect-${slotId}`);

                        applySlotClass(slotRect, data.status);

                        // Slot Text Label / Brand Name
                        const slotText = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                        slotText.setAttribute('x', slotX + (SLOT_SIZE / 2));
                        slotText.setAttribute('y', slotY + (SLOT_SIZE / 2) + 3);
                        slotText.setAttribute('fill', '#ffffff');
                        slotText.setAttribute('font-size', '8.5');
                        slotText.setAttribute('font-weight', '700');
                        slotText.setAttribute('text-anchor', 'middle');
                        slotText.setAttribute('pointer-events', 'none');
                        slotText.setAttribute('id', `text-${slotId}`);
                        slotText.textContent = data.company ? data.company.substring(0, 6) : `${r + 1}-${c + 1}`;

                        slotGroup.appendChild(slotRect);
                        slotGroup.appendChild(slotText);

                        slotGroup.addEventListener('click', (e) => {
                            e.stopPropagation();
                            selectSlot(slotId);
                        });

                        panelGroup.appendChild(slotGroup);
                    }
                }

                // Folding Lines Indicators
                if (['BASE', 'POSTERIOR', 'TAPA', 'SOLAPA'].includes(panel.id)) {
                    const fold = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                    fold.setAttribute('x1', px);
                    fold.setAttribute('y1', py);
                    fold.setAttribute('x2', px + panel.width);
                    fold.setAttribute('y2', py);
                    fold.setAttribute('class', 'fold-line');
                    panelGroup.appendChild(fold);
                }

                rootGroup.appendChild(panelGroup);
            });

            svg.appendChild(rootGroup);
        }

        /**
         * Update Slot Color Styling
         */
        function applySlotClass(rectElem, status) {
            rectElem.setAttribute('class', '');
            if (activeViewMode === 'COMMERCIAL') {
                if (status === 'AVAILABLE') rectElem.classList.add('slot-available');
                else if (status === 'RESERVED') rectElem.classList.add('slot-reserved');
                else if (status === 'SOLD') rectElem.classList.add('slot-sold');
            } else {
                rectElem.classList.add('slot-available');
            }

            if (rectElem.getAttribute('id') === `rect-${selectedSlotId}`) {
                rectElem.classList.add('slot-selected');
            }
        }

        /**
         * Select Slot and Populate Inspector Form
         */
        function selectSlot(slotId) {
            if (selectedSlotId) {
                const prev = document.getElementById(`rect-${selectedSlotId}`);
                if (prev) prev.classList.remove('slot-selected');
            }

            selectedSlotId = slotId;
            const data = slotsData[slotId];
            if (!data) return;

            const curr = document.getElementById(`rect-${slotId}`);
            if (curr) curr.classList.add('slot-selected');

            document.getElementById('no-slot-selected').classList.add('hidden');
            document.getElementById('slot-editor').classList.remove('hidden');

            document.getElementById('editor-zone-name').textContent = data.zoneName;
            document.getElementById('editor-slot-id').textContent = data.id;
            document.getElementById('editor-coords').textContent = `Fila ${data.row}, Col ${data.col}`;
            document.getElementById('input-company').value = data.company || '';
            document.getElementById('input-url').value = data.url || '';
            document.getElementById('input-price').value = data.price || standardPrice;
            document.getElementById('input-contact').value = data.contact || '';

            updateStatusButtonsUI(data.status);
            generateSingleQR(data.url || `https://bolso-qr.com/slot/${data.id}`);
        }

        function updateStatusButtonsUI(status) {
            document.querySelectorAll('.status-btn').forEach(btn => {
                if (btn.getAttribute('data-status') === status) {
                    btn.classList.add('ring-2', 'ring-white', 'ring-offset-2', 'ring-offset-slate-900');
                } else {
                    btn.classList.remove('ring-2', 'ring-white', 'ring-offset-2', 'ring-offset-slate-900');
                }
            });
        }

        function generateSingleQR(text) {
            const container = document.getElementById('qrcode-single-preview');
            container.innerHTML = '';
            new QRCode(container, {
                text: text,
                width: 100,
                height: 100,
                colorDark: "#0f172a",
                colorLight: "#ffffff",
                correctLevel: QRCode.CorrectLevel.H
            });
        }

        /**
         * Recalculate Financial Metrics & Breakdown
         */
        function recalculateMetrics() {
            let availableCount = 0;
            let reservedCount = 0;
            let soldCount = 0;

            let soldRevenue = 0;
            let reservedRevenue = 0;
            let potentialRevenue = 0;

            const zoneStats = {};

            Object.values(slotsData).forEach(s => {
                if (!zoneStats[s.zoneId]) {
                    zoneStats[s.zoneId] = { name: s.zoneName, total: 0, sold: 0, reserved: 0 };
                }
                zoneStats[s.zoneId].total++;

                const slotVal = parseFloat(s.price) || standardPrice;

                if (s.status === 'AVAILABLE') {
                    availableCount++;
                    potentialRevenue += slotVal;
                } else if (s.status === 'RESERVED') {
                    reservedCount++;
                    reservedRevenue += slotVal;
                    zoneStats[s.zoneId].reserved++;
                } else if (s.status === 'SOLD') {
                    soldCount++;
                    soldRevenue += slotVal;
                    zoneStats[s.zoneId].sold++;
                }
            });

            // Update Header Stats
            document.getElementById('hdr-available').textContent = availableCount;
            document.getElementById('hdr-reserved').textContent = reservedCount;
            document.getElementById('hdr-sold').textContent = soldCount;
            document.getElementById('hdr-revenue').textContent = `$${soldRevenue} USD`;

            // Update Financial Tab Stats
            document.getElementById('fin-total-sold').textContent = `$${soldRevenue} USD`;
            document.getElementById('fin-total-reserved').textContent = `$${reservedRevenue} USD`;
            document.getElementById('fin-total-potential').textContent = `$${potentialRevenue} USD`;
            document.getElementById('fin-grand-total').textContent = `$${soldRevenue + reservedRevenue + potentialRevenue} USD`;

            // Zone Breakdown Render
            const bdContainer = document.getElementById('fin-zones-breakdown');
            bdContainer.innerHTML = '';
            Object.keys(zoneStats).forEach(zKey => {
                const z = zoneStats[zKey];
                const soldPct = Math.round(((z.sold + z.reserved) / z.total) * 100);

                const row = document.createElement('div');
                row.className = 'flex flex-col gap-1 border-b border-slate-700/50 pb-1.5';
                row.innerHTML = `
                    <div class="flex justify-between font-semibold">
                        <span class="${zKey === 'BASE' ? 'text-sky-400' : 'text-slate-300'}">${z.name}</span>
                        <span class="text-slate-200">${z.sold + z.reserved}/${z.total} (${soldPct}%)</span>
                    </div>
                    <div class="w-full bg-slate-950 h-1.5 rounded-full overflow-hidden flex">
                        <div class="bg-emerald-500 h-full" style="width: ${(z.sold/z.total)*100}%"></div>
                        <div class="bg-amber-500 h-full" style="width: ${(z.reserved/z.total)*100}%"></div>
                    </div>
                `;
                bdContainer.appendChild(row);
            });

            renderActiveSponsorList();
        }

        /**
         * Render Active Sponsor Cards List
         */
        function renderActiveSponsorList() {
            const container = document.getElementById('active-sponsors-list');
            container.innerHTML = '';

            const activeSlots = Object.values(slotsData).filter(s => s.status !== 'AVAILABLE');

            if (activeSlots.length === 0) {
                container.innerHTML = `<p class="text-slate-500 text-center py-6">No hay reservas activas.</p>`;
                return;
            }

            activeSlots.forEach(s => {
                const card = document.createElement('div');
                card.className = 'bg-slate-800/80 hover:bg-slate-800 border border-slate-700/80 p-2.5 rounded-lg space-y-1 cursor-pointer transition';
                card.onclick = () => selectSlot(s.id);

                const isSold = s.status === 'SOLD';

                card.innerHTML = `
                    <div class="flex items-center justify-between">
                        <span class="font-bold text-slate-200">${s.company || 'Sin nombre'}</span>
                        <span class="px-1.5 py-0.5 text-[9px] font-bold rounded border ${isSold ? 'bg-emerald-950 border-emerald-800 text-emerald-400' : 'bg-amber-950 border-amber-800 text-amber-400'}">${isSold ? 'Vendido' : 'Reservado'}</span>
                    </div>
                    <div class="flex justify-between text-[10px] text-slate-400">
                        <span>${s.zoneName}</span>
                        <span class="font-mono text-indigo-300">${s.id}</span>
                    </div>
                `;
                container.appendChild(card);
            });
        }

        /**
         * Event Bindings
         */
        function bindEvents() {
            // Save Single Slot
            document.getElementById('btn-save-single').addEventListener('click', () => {
                if (!selectedSlotId) return;
                const d = slotsData[selectedSlotId];
                d.company = document.getElementById('input-company').value.trim();
                d.url = document.getElementById('input-url').value.trim();
                d.price = parseFloat(document.getElementById('input-price').value) || standardPrice;
                d.contact = document.getElementById('input-contact').value.trim();

                const textElem = document.getElementById(`text-${selectedSlotId}`);
                if (textElem) textElem.textContent = d.company ? d.company.substring(0, 6) : `${d.row}-${d.col}`;

                recalculateMetrics();
            });

            document.getElementById('btn-clear-single').addEventListener('click', () => {
                if (!selectedSlotId) return;
                const d = slotsData[selectedSlotId];
                d.status = 'AVAILABLE';
                d.company = '';
                d.url = '';
                d.contact = '';

                const rect = document.getElementById(`rect-${selectedSlotId}`);
                if (rect) applySlotClass(rect, 'AVAILABLE');

                selectSlot(selectedSlotId);
                recalculateMetrics();
            });

            document.querySelectorAll('.status-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    if (!selectedSlotId) return;
                    const st = btn.getAttribute('data-status');
                    slotsData[selectedSlotId].status = st;
                    updateStatusButtonsUI(st);

                    const rect = document.getElementById(`rect-${selectedSlotId}`);
                    if (rect) applySlotClass(rect, st);

                    recalculateMetrics();
                });
            });

            // Batch / Zone Assignment Operation
            document.getElementById('btn-apply-batch').addEventListener('click', () => {
                const zoneId = document.getElementById('batch-zone-select').value;
                const newStatus = document.getElementById('batch-status-select').value;
                const company = document.getElementById('batch-company-input').value.trim();
                const url = document.getElementById('batch-url-input').value.trim();
                const price = parseFloat(document.getElementById('batch-price-input').value) || standardPrice;

                Object.values(slotsData).forEach(s => {
                    if (s.zoneId === zoneId) {
                        s.status = newStatus;
                        if (newStatus !== 'AVAILABLE') {
                            s.company = company || s.company;
                            s.url = url || s.url;
                            s.price = price;
                        } else {
                            s.company = '';
                            s.url = '';
                        }

                        const rect = document.getElementById(`rect-${s.id}`);
                        if (rect) applySlotClass(rect, s.status);

                        const textElem = document.getElementById(`text-${s.id}`);
                        if (textElem) textElem.textContent = s.company ? s.company.substring(0, 6) : `${s.row}-${s.col}`;
                    }
                });

            });

                document.querySelectorAll('.panel-group').forEach(group => {
                    const zid = group.getAttribute('data-zone');
                    group.style.display = (zVal === 'ALL' || zid === zVal) ? 'block' : 'none';

            document.getElementById('file-import-json').addEventListener('change', (e) => {
                const file = e.target.files[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onload = (evt) => {
                    try {
                        slotsData = JSON.parse(evt.target.result);
                        renderSVGBlueprint();
                        recalculateMetrics();
                        alert('¡Datos cargados exitosamente!');
                    } catch (err) {
                        alert('Error al importar archivo JSON');
                    }
                };
                reader.readText(file);
            });
        }
